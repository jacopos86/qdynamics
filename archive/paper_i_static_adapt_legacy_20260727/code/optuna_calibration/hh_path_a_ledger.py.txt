#!/usr/bin/env python3
"""Regenerate the neutral Path A HH candidate ledger."""
from __future__ import annotations

import argparse
import sys
from dataclasses import dataclass
from datetime import datetime, timezone
from pathlib import Path
from typing import Sequence

REPO_ROOT = Path(__file__).resolve().parents[2]
if str(REPO_ROOT) not in sys.path:
    sys.path.insert(0, str(REPO_ROOT))

from pipelines.exact_bench.hh_cost_energy_optuna import _PUBLIC_ANCHOR_COMMAND
from pipelines.exact_bench.hh_spsa_powell_handoff_refinement import SEED_LIBRARY

DEFAULT_OUTPUT_PATH = REPO_ROOT / "pipelines/exact_bench/HH_PATH_A_LEDGER.md"
DEFAULT_CANDIDATE_ORDER = (
    "public_anchor_apr05",
    "current_98",
    "current_118",
    "legacy_75",
    "legacy_81",
)
PUBLIC_ANCHOR_DELTA_E_ABS = 1.0822209459e-04
PUBLIC_ANCHOR_COMPILED_2Q = 218
PUBLIC_ANCHOR_NOTE = (
    "Public direct-route SPSA anchor used by the Path A Optuna preset library; "
    "kept here as recovery metadata rather than a promoted verdict."
)
NEUTRAL_PROVENANCE_NOTES = {
    "current_98": "Current-route bridge-focused Powell artifact pinned as a handoff seed in `hh_spsa_powell_handoff_refinement.py`.",
    "current_118": "Current-route bridge-focused Powell artifact pinned as a handoff seed in `hh_spsa_powell_handoff_refinement.py`.",
    "legacy_75": "Legacy-focused Powell artifact pinned as a handoff seed in `hh_spsa_powell_handoff_refinement.py`.",
    "legacy_81": "Frozen March 22 legacy oracle pinned as a Path A seed and motif reference.",
}


@dataclass(frozen=True)
class LedgerCandidate:
    label: str
    delta_e_abs: float
    compiled_count_2q: int
    compiled_depth: int | None
    json_paths: tuple[Path, ...]
    command_log_paths: tuple[Path, ...]
    provenance_note: str


# Built Math: p_local = count(existing pinned paths) / count(all pinned paths).
def _path_status(paths: Sequence[Path]) -> str:
    existing = sum(1 for path in paths if Path(path).exists())
    total = len(list(paths))
    if total == 0:
        return "no paths recorded"
    if existing == total:
        return "all recorded paths present locally"
    if existing == 0:
        return "recorded paths absent locally in this checkout"
    return f"{existing}/{total} recorded paths present locally"


def _repo_rel(path: Path) -> str:
    try:
        return str(Path(path).resolve().relative_to(REPO_ROOT))
    except Exception:
        return str(path)


def _case_dir_from_result_json(result_json: Path) -> Path:
    return Path(result_json).parent.parent


def _compile_json_from_result_json(result_json: Path) -> Path:
    return Path(result_json).parent / "compile_scout_fake_marrakesh.json"


def _default_command_log_paths(case_dir: Path) -> tuple[Path, ...]:
    return (
        Path(case_dir) / "logs" / "command.sh",
        Path(case_dir) / "logs" / "stdout.log",
        Path(case_dir) / "logs" / "stderr.log",
    )


def _seed_candidate(name: str) -> LedgerCandidate:
    spec = SEED_LIBRARY[str(name)]
    result_json = Path(spec.input_json)
    case_dir = _case_dir_from_result_json(result_json)
    return LedgerCandidate(
        label=str(name),
        delta_e_abs=float(spec.expected_delta_abs),
        compiled_count_2q=int(spec.expected_two_qubit_count),
        compiled_depth=int(spec.expected_depth),
        json_paths=(result_json, _compile_json_from_result_json(result_json)),
        command_log_paths=_default_command_log_paths(case_dir),
        provenance_note=str(NEUTRAL_PROVENANCE_NOTES.get(str(name), spec.notes)),
    )


def _public_anchor_candidate() -> LedgerCandidate:
    artifact_dir = Path(_PUBLIC_ANCHOR_COMMAND).parent.parent
    result_json = artifact_dir / "json" / "result.json"
    return LedgerCandidate(
        label="public_anchor_apr05",
        delta_e_abs=float(PUBLIC_ANCHOR_DELTA_E_ABS),
        compiled_count_2q=int(PUBLIC_ANCHOR_COMPILED_2Q),
        compiled_depth=None,
        json_paths=(result_json,),
        command_log_paths=(
            Path(_PUBLIC_ANCHOR_COMMAND),
            artifact_dir / "logs" / "stdout.log",
            artifact_dir / "logs" / "stderr.log",
        ),
        provenance_note=PUBLIC_ANCHOR_NOTE,
    )


def build_candidate_rows() -> list[LedgerCandidate]:
    return [
        _public_anchor_candidate(),
        _seed_candidate("current_98"),
        _seed_candidate("current_118"),
        _seed_candidate("legacy_75"),
        _seed_candidate("legacy_81"),
    ]


def render_ledger(candidates: Sequence[LedgerCandidate], generated_utc: str | None = None) -> str:
    stamp = str(generated_utc or datetime.now(timezone.utc).isoformat())
    lines = [
        "# HH Path A Candidate Ledger",
        "",
        f"Auto-generated by `python -m pipelines.exact_bench.hh_path_a_ledger` on `{stamp}`.",
        "",
        "Path A is a proposal lane and replay aid. This ledger records candidate runs/results as neutral candidate rows only.",
        "",
        "## Scope and maintenance rule",
        "",
        "- Scope: Path A only (`hh_cost_energy_optuna.py`, `hh_spsa_powell_handoff_refinement.py`, `hh_math_top2_monitor.py`, `hh_path_a_ledger.py`)",
        "- Purpose: keep candidate metadata recoverable if a later confirmation/promotion decision is made",
        "- Update rule: regenerate this file from repo root whenever Path A pins a new candidate path or metric in code",
        "- Non-goal: this file does **not** rank or promote candidates",
        "",
        "## Current checkout note",
        "",
        "- Ledger rows preserve code-pinned recovery metadata even when the underlying artifact paths are absent locally.",
        "- Path status below is local to the current checkout only.",
        "",
        "## Candidate rows",
        "",
    ]
    for candidate in candidates:
        combined_paths = [*candidate.json_paths, *candidate.command_log_paths]
        lines.extend(
            [
                f"### {candidate.label}",
                f"- Short label / run tag: `{candidate.label}`",
                f"- `|ΔE|`: `{candidate.delta_e_abs:.16g}`",
                f"- Compiled 2Q: `{candidate.compiled_count_2q}`",
                (
                    f"- Compiled depth: `{candidate.compiled_depth}`"
                    if candidate.compiled_depth is not None
                    else "- Compiled depth: `—` (not pinned in current Path A code)"
                ),
                f"- Path status: {_path_status(combined_paths)}",
                "- JSON path(s):",
            ]
        )
        for path in candidate.json_paths:
            lines.append(f"  - `{_repo_rel(path)}`")
        lines.append("- Command/log path(s):")
        for path in candidate.command_log_paths:
            lines.append(f"  - `{_repo_rel(path)}`")
        lines.append(f"- Brief provenance note: {candidate.provenance_note}")
        lines.append("")
    return "\n".join(lines).rstrip() + "\n"


def build_parser() -> argparse.ArgumentParser:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--output-path", type=Path, default=DEFAULT_OUTPUT_PATH)
    return parser


def main(argv: Sequence[str] | None = None) -> Path:
    args = build_parser().parse_args(list(argv) if argv is not None else None)
    output_path = Path(args.output_path)
    output_path.parent.mkdir(parents=True, exist_ok=True)
    rendered = render_ledger(build_candidate_rows())
    output_path.write_text(rendered, encoding="utf-8")
    print(f"Wrote Path A ledger: {output_path}")
    return output_path


if __name__ == "__main__":
    main()
