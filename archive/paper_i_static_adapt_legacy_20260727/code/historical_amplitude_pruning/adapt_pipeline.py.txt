    def _refresh_prune_metadata_stagnation(
        metadata_rows: Sequence[ScaffoldCoordinateMetadata],
        theta_logical_now: np.ndarray,
    ) -> list[ScaffoldCoordinateMetadata]:
        theta_vals = np.asarray(theta_logical_now, dtype=float).reshape(-1)
        theta_scale = float(max(np.median(np.abs(theta_vals)) if theta_vals.size > 0 else 0.0, 1e-12))
        out: list[ScaffoldCoordinateMetadata] = []
        for idx, meta in enumerate(metadata_rows):
            theta_raw = float(theta_vals[idx]) if idx < int(theta_vals.size) else 0.0
            theta_abs = float(abs((theta_raw + math.pi) % (2.0 * math.pi) - math.pi))
            obs_count_prev = int(max(0, int(meta.amplitude_observation_count)))
            previous_abs = float(meta.current_abs_theta) if obs_count_prev > 0 else float(theta_abs)
            recent_abs_motion = float(abs(theta_abs - previous_abs))
            cumulative_abs_motion = float(max(0.0, float(meta.cumulative_abs_motion)) + recent_abs_motion)
            peak_abs = float(max(float(meta.peak_abs_theta), theta_abs, previous_abs))
            out.append(
                ScaffoldCoordinateMetadata(
                    **{
                        **meta.__dict__,
                        "stagnation_score": float(max(0.0, 1.0 - theta_abs / theta_scale)),
                        "current_abs_theta": float(theta_abs),
                        "previous_abs_theta": float(previous_abs),
                        "peak_abs_theta": float(peak_abs),
                        "amplitude_observation_count": int(obs_count_prev + 1),
                        "recent_abs_motion": float(recent_abs_motion),
                        "cumulative_abs_motion": float(cumulative_abs_motion),
                    }
                )
            )
        return out
