def _problem_context(*, family: str = "hh") -> ResolvedProblemContext:
    hamiltonian = object()
    request = ProblemRequest(
        problem_key=str(family),
        num_sites=2,
        t=1.0,
        u=2.0,
        dv=0.0,
        omega0=1.0,
        g_ep=0.5,
        n_ph_max=1,
        boson_encoding="binary",
        ordering="blocked",
        boundary="open",
    )
    return ResolvedProblemContext(
        family_key=str(family),
        request=request,
        layout=RegisterLayoutSpec(
            total_qubits=6,
            fermion_qubits=4,
            boson_qubits=2,
            ordering="blocked",
            boson_encoding="binary",
            blocks=(),
        ),
        hamiltonian=hamiltonian,
        sector=SectorSelection(
            label="test",
            comparison_space_label="test",
            constraints=(FixedCountConstraint(quantity="fermions", value=2),),
            num_particles=(1, 1),
        ),
        reference_state=ReferenceStateSpec(
            kind="test",
            source_label="test",
            state_kind="statevector",
            build_state=lambda: np.asarray([1.0], dtype=complex),
        ),
        exact_target=ExactTargetSpec(
            kind="test",
            comparison_space_label="test",
            resolve_energy=lambda **_kwargs: -1.0,
            exact_state_policy="fallback",
            build_fallback_anchor_state=lambda: np.asarray([1.0], dtype=complex),
            fallback_policy="test",
        ),
        default_controller_profile="test",
        default_continuation_mode="phase3_v1",
        admissible_pool_keys=("full_meta",),
        default_pool_key="full_meta",
        default_pool_resolution_scope="test",
        default_sector_label="test",
        default_reference_label="test",
        exact_target_label="test",
        exact_comparison_space_label="test",
        default_num_particles=(1, 1),
    )

def test_typed_runner_exposes_exact_structural_resume(monkeypatch, tmp_path) -> None:
    source_path = tmp_path / "round7-current.json"
    source_path.write_text("{}", encoding="utf-8")
    source = SimpleNamespace(
        artifact_json=source_path.resolve(),
        artifact_sha256="a" * 64,
        payload={},
        runtime_input=SimpleNamespace(
            psi_ref=np.asarray([1.0 + 0.0j] + [0.0 + 0.0j] * 63)
        ),
    )
    monkeypatch.setattr(
        paper_i_runner,
        "load_static_resume_source",
        lambda *args, **kwargs: source,
    )
    monkeypatch.setattr(
        paper_i_runner,
        "validate_static_hh_resume_source",
        lambda *args, **kwargs: {"settings_match": True},
    )
    monkeypatch.setattr(
        paper_i_runner,
        "build_resume_import_summary",
        lambda *args, **kwargs: {
            "source_controller_round": 7,
            "source_ansatz_depth": 12,
        },
    )

    invocation = build_paper_i_route_a_invocation(
        _problem_context(),
        run_config=PaperISnakeRunConfig(
            max_adapt_iterations=23,
            resume_scaffold_json=source_path,
        ),
    )

    assert invocation.run_kwargs["adapt_resume_source"] is source
    assert invocation.run_kwargs["adapt_resume_scaffold_json"] == source_path.resolve()
    assert invocation.run_kwargs["adapt_ref_base_depth"] == 12
    assert invocation.run_kwargs["max_depth"] == 23
    assert invocation.manifest["resume"] == {
        "enabled": True,
        "source_json": str(source_path.resolve()),
        "source_sha256": "a" * 64,
        "source_controller_round": 7,
        "source_ansatz_depth": 12,
        "requested_additional_controller_rounds": 23,
    }

def _real_hh_context(*, n_ph_max: int = 1) -> ResolvedProblemContext:
    request = ProblemRequest(
        problem_key="hh",
        num_sites=2,
        t=1.0,
        u=0.5,
        dv=0.0,
        omega0=1.0,
        g_ep=0.2,

        include_zero_point=True,
