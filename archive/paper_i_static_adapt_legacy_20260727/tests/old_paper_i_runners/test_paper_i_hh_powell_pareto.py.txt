def test_powell_harness_reports_fixed_fermion_sector_probability() -> None:

    good = np.zeros(256, dtype=complex)
    good[int("00000101", 2)] = 1.0
    leaking = np.zeros(256, dtype=complex)
    leaking[int("00000001", 2)] = 1.0

    good_diag = harness._fixed_fermion_sector_diagnostics(good, context)
    leak_diag = harness._fixed_fermion_sector_diagnostics(leaking, context)

    assert good_diag["status"] == "ok"
    assert good_diag["probability"] == pytest.approx(1.0)
    assert good_diag["leak_probability"] == pytest.approx(0.0)
    assert leak_diag["probability"] == pytest.approx(0.0)
    assert leak_diag["leak_probability"] == pytest.approx(1.0)

def test_powell_harness_surfaces_child_padding_population_telemetry(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    monkeypatch.setattr(
        harness,
        "snake_algorithmic_work_from_payload",
        lambda *_args, **_kwargs: (
            {
                "S_alg": 0.0,
                "S_alg_status": "ok",
                "S_alg_work_scope": "terminal",
            },
            {},
        ),
    )
    payload = {
        "success": True,
        "operators": [],
        "logical_optimal_point": [],
        "continuation": {
            "runtime_split_summary": {
                "route_a_funnel_last_round": {
                    "globally_unique_eligible_child_identity_count": 7,
                    "child_padding_filter": {
                        "policy_effective": (
                            harness.ROUTE_A_CHILD_PADDING_HARD_FILTER_NPH2_V1

                        "input_record_count": 10,
                        "remaining_record_count": 8,
                        "rejected_record_count": 2,
                        "rejected_identity_count": 1,
                        "rejected_identities": [{"identity": "pauli:test"}],

                    "joint_selector_summary": {
                        "child_phase2_rank_feasible_count": 6,
                        "rank_prefilter_rejection_count": 2,
                        "rank_feasibility_prefilter": {
                            "policy": "rank_feasible_fill_v1",
                            "input_record_count": 8,

        },
    }

    summary = harness._result_summary(

    assert selector["child_phase2_rank_feasible_count"] == 6
    assert selector["rank_prefilter_rejection_count"] == 2
    assert selector["rank_feasibility_prefilter"]["policy"] == (

def test_powell_harness_worker_parity_report() -> None:
    summary = {
        "energy": -1.0,
        "elapsed_s": 10.0,
        "operator_path_hash": "same",
        "logical_optimal_point": [0.1, 0.2],
        "query_work_total": 123.0,
    }
    left = {
        "status": "complete",
        "scientific_settings_hash": "settings",
        "execution_settings": {"gradient_workers": 1},
        "process_cpu_s": 9.0,
        "host_loadavg_start": [0.5, 0.4, 0.3],
        "summary": dict(summary),
    }
    right = {
        "status": "complete",
        "scientific_settings_hash": "settings",
        "execution_settings": {"gradient_workers": 8},
        "process_cpu_s": 18.0,
        "host_loadavg_start": [0.6, 0.5, 0.4],
        "summary": {**summary, "elapsed_s": 4.0},
    }

    report = harness.compare_runs(left, right, atol=1e-12)

    assert report["status"] == "pass"
    assert report["parallel_speedup"] == pytest.approx(2.5)
    assert report["parallel_to_serial_process_cpu_ratio"] == pytest.approx(2.0)
    assert all(report["checks"].values())

def test_powell_harness_uses_winning_branch_s_alg_for_beam_query_work(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    def _accounting(payload, *, scope, source_label):
        nfev = float(payload["nfev_total"])
        return (
            {
                "S_alg": nfev + 20.0,
                "S_alg_status": "ok",
                "S_alg_work_scope": "winner_lineage_terminal",
                "S_alg_N_H_outer_eval": nfev - 10.0,
                "S_alg_N_H_refit_eval": 10.0,
                "S_alg_N_grad_probe": 5.0,
                "S_alg_N_metric_probe": 15.0,
            },

                "S_beam_search_total": nfev + 80.0,

                    "N_H_outer_eval": nfev - 10.0,
                    "N_H_refit_eval": 10.0,
                    "N_grad": 30.0,
                    "N_metric": 50.0,
                },

    assert low["query_work_components"] == {
        "N_H_outer": 90.0,
        "N_H_refit": 10.0,
        "N_grad": 5.0,
        "N_metric": 15.0,
    }
    assert low["all_branch_query_work_total_diagnostic"] == pytest.approx(180.0)
    assert high["all_branch_query_work_total_diagnostic"] == pytest.approx(280.0)
    assert low["all_branch_query_work_scope_diagnostic"] == (
        "all_expanded_scored_branches"
    )

@pytest.mark.parametrize(
    ("mutation", "message"),
    [
        ("wrong_label", "order mismatch"),
        ("missing_terms", "no serialized terms"),
        ("bad_pauli", "invalid characters"),
        ("inconsistent_nq", "Inconsistent terminal Pauli widths"),
        ("nonfinite_coefficient", "nonfinite coefficient"),
    ],
)
def test_terminal_internal_graph_proxy_fails_closed(
    mutation: str,
    message: str,
) -> None:
    payload = _terminal_proxy_payload()
    metadata = payload["continuation"]["selected_generator_metadata"]
    if mutation == "wrong_label":
        metadata[1]["candidate_label"] = "wrong"
    elif mutation == "missing_terms":
        metadata[1]["compile_metadata"]["serialized_terms_exyz"] = []
    elif mutation == "bad_pauli":
        metadata[1]["compile_metadata"]["serialized_terms_exyz"][0][
            "pauli_exyz"
        ] = "xxqe"
    elif mutation == "inconsistent_nq":
        metadata[1]["compile_metadata"]["serialized_terms_exyz"][0]["nq"] = 5
    elif mutation == "nonfinite_coefficient":
        metadata[1]["compile_metadata"]["serialized_terms_exyz"][0][
            "coeff_re"
        ] = float("nan")
    with pytest.raises(ValueError, match=message):
        harness.terminal_internal_graph_proxy_from_payload(payload)
