def test_all_infeasible_novelty_fallback_telemetry_counts_ordered_rounds() -> None:
    receipt = (
        adapt_pipeline._all_energy_models_infeasible_novelty_fallback_telemetry(
            [
                {
                    "depth": 2,
                    "depth_cumulative": 12,
                    "selected_op": "a",
                    "all_energy_models_infeasible_novelty_fallback_fired": True,
                    "route_a_geometry_expansion_reason": (
                        "all_whitened_energy_models_infeasible"
                    ),
                },
                {
                    "depth": 3,
                    "depth_cumulative": 13,
                    "selected_op": "b",
                    "selected_feature_rows": [
                        {
                            "route_a_geometry_expansion_mode": (
                                "collective_span_novelty_over_cost_v1"
                            ),
                            "route_a_geometry_expansion_reason": (
                                "all_whitened_energy_models_infeasible"
                            ),
                            "route_a_geometry_expansion_novelty_query_charge": 0,
                        }
                    ],
                },
            ],
            phase2_gram_novelty_policy=GRAM_NOVELTY_POLICY_FALLBACK_ONLY_V1,
            phase3_gram_novelty_policy=GRAM_NOVELTY_POLICY_FALLBACK_ONLY_V1,
            hardware_cost_normalization_mode=(
                "family_robust_symmetric_arctan_v1"
            ),
        )
    )
    assert receipt["enabled"] is True
    assert receipt["fired"] is True
    assert receipt["activation_count"] == 2
    assert receipt["controller_rounds"] == [12, 13]
    assert receipt["selected_operators"] == ["a", "b"]
    assert receipt["reason_counts"] == {
        "all_whitened_energy_models_infeasible": 2
    }
    assert receipt["query_charge_total"] == 0
