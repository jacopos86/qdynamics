def _rows(tsv_text: str) -> list[dict[str, str]]:
    return list(csv.DictReader(tsv_text.splitlines(), delimiter="\t"))


def _write_snake_candidate_manifest(tmp_path: Path, *, target_case_id: str = "hh_L2_nph3_clean_weak") -> Path:
    source_case_id = target_case_id.replace("_nph3_", "_nph2_")
    audit_path = tmp_path / "source_audit.json"
    audit_path.write_text('{"schema":"paper_i_fixed_accuracy_table_audit_v1"}\n', encoding="utf-8")
    manifest = {
        "schema": audit_escalation.CANDIDATE_SCHEMA,
        "threshold": 2e-4,
        "target_profile": "paper_i_phys_v1",
        "source_stage": "nph2_ref4_screen",
        "next_stage": "nph3_ref4_escalation",
        "candidates": [
            {
                "candidate_key": f"snake|hh|SNAKE|weak|{target_case_id}|static_family_native_adapt_phase3",
                "lane": "snake",
                "family": "hh",
                "method": "SNAKE",
                "regime": "weak",
                "algorithm_id": "static_family_native_adapt_phase3",
                "source_case_id": source_case_id,
                "source_stage": "nph2_ref4_screen",
                "source_n_ph_work": 2,
                "source_n_ph_ref": 4,
                "source_status": "failed",
                "source_threshold_status": "failed",
                "source_record_id": f"static_table__hh__{source_case_id}__static_family_native_adapt_phase3",
                "source_payload_path": "raw_outputs/example/live_snake.json",
                "source_payload_sha256": "abc123",
                "source_audit_json": str(audit_path),
                "source_audit_sha256": "def456",
                "source_target_profile": "paper_i_phys_v1",
                "source_threshold": 2e-4,
                "next_stage": "nph3_ref4_escalation",
                "next_stage_case_id": target_case_id,
                "target_n_ph_work": 3,
                "target_n_ph_ref": 4,
                "target_suite_profile": "paper_i_clean_nph3_ref4_v1",
                "escalation_reason": "completed_failed_phonon_ladder_row",
            }
        ],
    }
    path = tmp_path / "candidates.json"
    path.write_text(json.dumps(manifest, indent=2, sort_keys=True) + "\n", encoding="utf-8")
    return path


def _write_submit(tmp_path: Path, *, records_rel: str, ids_rel: str) -> Path:
    submit = tmp_path / "submit.sub"
    submit.write_text(
        "\n".join(
            [
                "transfer_input_files = .",
                f'environment = "PHASE3_RECORDS_PATH={records_rel}"',
                f"queue record_id from {ids_rel}",
                "",
            ]
        ),
        encoding="utf-8",
    )
    return submit
