from __future__ import annotations

import json
import sys
from pathlib import Path
from types import SimpleNamespace

REPO_ROOT = Path(__file__).resolve().parents[2]
if str(REPO_ROOT) not in sys.path:
    sys.path.insert(0, str(REPO_ROOT))

from chtc.phase3_optuna import hh_same_cutoff_overnight as overnight


def _arg_value(args: tuple[str, ...], flag: str) -> str:
    return args[args.index(flag) + 1]


def _write_current_best(tmp_path: Path, lane: overnight.LaneConfig, result: dict, *, trial: int = 0) -> None:
    paths = overnight.lane_paths(tmp_path, lane.lane_id)
    paths["progress_dir"].mkdir(parents=True, exist_ok=True)
    payload = {
        "schema": "phase3_progress_current_v1",
        "best_trial_number": trial,
        "best_value": 0.0,
        "best_user_attrs": {"results": {lane.benchmark_id: result}},
    }
    (paths["progress_dir"] / "current_best.json").write_text(json.dumps(payload), encoding="utf-8")


def _write_complete_event(tmp_path: Path, lane: overnight.LaneConfig, *, trial: int = 0, state: str = "COMPLETE") -> None:
    paths = overnight.lane_paths(tmp_path, lane.lane_id)
    paths["progress_dir"].mkdir(parents=True, exist_ok=True)
    event = {"event": "trial_completed", "trial_number": trial, "state": state}
    with (paths["progress_dir"] / "trial_events.jsonl").open("a", encoding="utf-8") as fh:
        fh.write(json.dumps(event) + "\n")


def _mark_running_owned(state: dict, tmp_path: Path, lane: overnight.LaneConfig, *, pid: int) -> None:
    command = overnight.build_worker_command(lane=lane, run_root=tmp_path, n_trials=3, poll_seconds=0.1)
    state["lanes"][lane.lane_id].update({"state": "running", "pid": pid, "command": command})
    state.setdefault("owned_processes", {})[str(pid)] = {
        "lane_id": lane.lane_id,
        "pid": pid,
        "command": command,
        "run_root": str(tmp_path),
    }

def test_launch_pending_lanes_uses_conservative_one_lane_even_when_two_allowed(monkeypatch, tmp_path: Path) -> None:
    launched: list[list[str]] = []

    class FakePopen:
        def __init__(self, command, **_kwargs):
            self.command = list(command)
            self.pid = 4242 + len(launched)
            launched.append(list(command))

        def poll(self):
            return None

    monkeypatch.setattr(overnight.subprocess, "Popen", FakePopen)
    state = overnight.initial_state(tmp_path, max_local_processes=2, monitor_seconds=60)
    processes = {}

    overnight._launch_pending_lanes(
        state,
        tmp_path,
        processes,
        max_local_processes=2,
        n_trials=3,
        poll_seconds=0.1,
        inner_optimizer="SPSA",
        phase3_runtime_split_mode="off",
        adapt_max_depth=48,
    )

    assert list(processes) == ["weak_weak"]
    assert state["lanes"]["weak_weak"]["state"] == "running"
    assert state["lanes"]["medium"]["state"] == "pending"
    assert state["concurrency_policy"]["mode"] == "conservative_one_lane_weak_first"
    assert state["concurrency_policy"]["effective_lane_slots"] == 1
    assert launched[0][launched[0].index("--lane") + 1] == "weak_weak"


def test_scoreboard_extracts_current_best_hit(tmp_path: Path) -> None:
    lane = overnight.LANE_BY_ID["weak_weak"]
    paths = overnight.lane_paths(tmp_path, lane.lane_id)
    paths["progress_dir"].mkdir(parents=True)
    result_json = tmp_path / "lanes" / lane.lane_id / "optuna" / "trial_0003" / "result.json"
    payload = {
        "schema": "phase3_progress_current_v1",
        "best_trial_number": 3,
        "best_value": 0.0,
        "best_user_attrs": {
            "results": {
                lane.benchmark_id: {
                    "success": True,
                    "abs_delta_e": 1.5e-4,
                    "abs_delta_e_same_cutoff": 1.5e-4,
                    "abs_delta_e_reference": None,
                    "depth_2q": 52,
                    "circuit_depth": 2920,
                    "result_json": str(result_json),
                }
            }
        },
    }
    (paths["progress_dir"] / "current_best.json").write_text(json.dumps(payload), encoding="utf-8")
    _write_complete_event(tmp_path, lane, trial=3)
    state = overnight.initial_state(tmp_path, max_local_processes=2, monitor_seconds=60)
    state["lanes"][lane.lane_id]["state"] = "running"

    metrics = overnight.lane_metrics(tmp_path, lane)
    rendered = overnight.render_scoreboard(tmp_path, state)

    assert metrics["trial"] == 3
    assert metrics["hit"] is True
    assert "1.500e-04" in rendered
    assert "52/2920" in rendered
    assert "hit" in rendered
    assert "n/a (same cutoff)" in rendered

def test_background_restart_preserves_existing_lane_state(monkeypatch, tmp_path: Path) -> None:
    class FakePopen:
        def __init__(self, command, **_kwargs):
            self.command = list(command)
            self.pid = 77777

    monkeypatch.setattr(overnight.subprocess, "Popen", FakePopen)
    state = overnight.initial_state(tmp_path, max_local_processes=2, monitor_seconds=60)
    state["lanes"]["weak_weak"].update({"state": "running", "pid": 12345})
    overnight.save_state(tmp_path, state)

    args = SimpleNamespace(
        run_root=tmp_path,
        max_local_processes=2,
        monitor_seconds=60.0,
        poll_seconds=0.1,
        n_trials=3,
        inner_optimizer="SPSA",
        phase3_runtime_split_mode="off",
        adapt_max_depth=48,
    )
    overnight.run_background(args)

    reloaded = overnight.load_state(tmp_path)
    assert reloaded is not None
    assert reloaded["monitor_pid"] == 77777
    assert reloaded["lanes"]["weak_weak"]["state"] == "running"
    assert reloaded["lanes"]["weak_weak"]["pid"] == 12345
    assert reloaded["lanes"]["medium"]["state"] == "pending"


def test_no_kill_without_owned_processes_even_on_hit(monkeypatch, tmp_path: Path) -> None:
    lane = overnight.LANE_BY_ID["weak_weak"]
    _write_current_best(
        tmp_path,
        lane,
        {"success": True, "abs_delta_e_same_cutoff": 1.0e-4, "result_json": str(tmp_path / "result.json")},
    )
    _write_complete_event(tmp_path, lane)
    state = overnight.initial_state(tmp_path, max_local_processes=2, monitor_seconds=60)
    state["lanes"][lane.lane_id].update(
        {
            "state": "running",
            "pid": 55555,
            "command": overnight.build_worker_command(lane=lane, run_root=tmp_path, n_trials=3, poll_seconds=0.1),
        }
    )
    killed: list[tuple[int, int]] = []
    monkeypatch.setattr(overnight.os, "killpg", lambda pid, sig: killed.append((pid, sig)))

    overnight._poll_lanes(state, tmp_path, {})

    assert killed == []
    assert state["lanes"][lane.lane_id]["state"] == "running"
    assert state["lanes"][lane.lane_id]["stop_pending_reason"] == "same_cutoff_hit"
    assert state["stopped_jobs"][0]["status"] == "not_stopped_unverified_ownership"
    assert state["blockers"][0]["reason"] == "ownership_not_proven_for_stop"


def test_status_is_read_only_and_does_not_terminate(monkeypatch, tmp_path: Path) -> None:
    lane = overnight.LANE_BY_ID["weak_weak"]
    _write_current_best(tmp_path, lane, {"success": True, "abs_delta_e_same_cutoff": 1.0e-4})
    _write_complete_event(tmp_path, lane)
    state = overnight.initial_state(tmp_path, max_local_processes=2, monitor_seconds=60)
    _mark_running_owned(state, tmp_path, lane, pid=55555)
    overnight.save_state(tmp_path, state)

    def fail_killpg(_pid: int, _sig: int) -> None:
        raise AssertionError("status must not terminate")

    monkeypatch.setattr(overnight.os, "killpg", fail_killpg)
    monkeypatch.setattr(overnight, "_ps_process_rows", lambda: [])
    monkeypatch.setattr(overnight, "write_scoreboard", lambda *_args, **_kwargs: None)

    overnight.run_status(SimpleNamespace(run_root=tmp_path))

    reloaded = overnight.load_state(tmp_path)
    assert reloaded is not None
    assert reloaded["lanes"][lane.lane_id]["state"] == "running"
    assert reloaded["lanes"][lane.lane_id]["last_metrics"]["hit"] is True
    assert reloaded["stopped_jobs"] == []


def test_first_hit_stops_other_verified_owned_lanes(monkeypatch, tmp_path: Path) -> None:
    weak = overnight.LANE_BY_ID["weak_weak"]
    medium = overnight.LANE_BY_ID["medium"]
    _write_current_best(tmp_path, weak, {"success": True, "abs_delta_e_same_cutoff": 1.0e-4}, trial=2)
    _write_complete_event(tmp_path, weak, trial=2)
    state = overnight.initial_state(tmp_path, max_local_processes=2, monitor_seconds=60)
    _mark_running_owned(state, tmp_path, weak, pid=55555)
    _mark_running_owned(state, tmp_path, medium, pid=66666)
    killed: list[tuple[int, int]] = []
    monkeypatch.setattr(overnight.os, "killpg", lambda pid, sig: killed.append((pid, sig)))

    overnight._poll_lanes(state, tmp_path, {})

    assert killed == [(55555, overnight.signal.SIGTERM), (66666, overnight.signal.SIGTERM)]
    assert state["lanes"][weak.lane_id]["state"] == "hit"
    assert state["lanes"][medium.lane_id]["state"] == "blocked"
    assert state["lanes"][medium.lane_id]["stop_reason"] == "obsolete_after_hit"
    assert state["stop_launching"] is True
    assert [row["reason"] for row in state["stopped_jobs"]] == ["same_cutoff_hit", "obsolete_after_hit"]

def test_energy_reference_fallback_can_validate_hit(tmp_path: Path) -> None:
    lane = overnight.LANE_BY_ID["weak_weak"]
    _write_current_best(
        tmp_path,
        lane,
        {"success": True, "energy": -1.00015, "same_cutoff_exact_gs_energy": -1.00000},
    )
    _write_complete_event(tmp_path, lane)

    metrics = overnight.lane_metrics(tmp_path, lane)

    assert metrics["hit"] is True
    assert abs(metrics["same_cutoff_delta_e"] - 0.00015) < 1e-12
    assert metrics["hit_validation"]["delta_source"] == "abs_energy_minus_same_cutoff_exact_gs_energy"
