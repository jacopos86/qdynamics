def _policy(**overrides):
    values = {
        "macro_phase1_cap": 48,
        "macro_phase2_cap": 24,
        "child_phase1_cap": 64,
        "child_phase2_cap": 32,
        "batch_search_pool_size": 10,
        "batch_size_cap": 2,
        "lambda_add": 0.1,
        "batch_mode": "combinatorial",
    }
    values.update(overrides)
    return route_optuna.RouteAPolicy(**values)


def _write_json(path: Path, payload: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload), encoding="utf-8")


def test_policy_covers_all_four_shortlists_and_excludes_zero_additivity() -> None:
    policy = _policy(
        macro_phase1_cap=route_optuna.ALL,
        macro_phase2_cap=route_optuna.ALL,
        child_phase1_cap=route_optuna.ALL,
        child_phase2_cap=route_optuna.ALL,
        batch_search_pool_size=route_optuna.ALL,
    )

    assert policy.as_dict() == {
        "macro_phase1_cap": "all",
        "macro_phase2_cap": "all",
        "child_phase1_cap": "all",
        "child_phase2_cap": "all",
        "batch_search_pool_size": "all",
        "batch_size_cap": 2,
        "lambda_add": 0.1,
        "batch_mode": "combinatorial",
    }
    assert policy.resolved_caps() == {
        "macro_phase1_cap": route_optuna.UNBOUNDED_SHORTLIST_CAP,
        "macro_phase2_cap": route_optuna.UNBOUNDED_SHORTLIST_CAP,
        "child_phase1_cap": route_optuna.UNBOUNDED_SHORTLIST_CAP,
        "child_phase2_cap": route_optuna.UNBOUNDED_SHORTLIST_CAP,
        "batch_search_pool_size": 0,
    }
    with pytest.raises(ValueError, match="unpenalized zero route is excluded"):
        _policy(lambda_add=0.0)

def test_plan_wave_writes_one_policy_across_all_six_regimes(tmp_path: Path) -> None:
    database = tmp_path / "study.sqlite3"
    wave_root = tmp_path / "wave"

    wave = route_optuna.plan_wave(
        database=database,
        study_name="route-a-test",
        output_dir=wave_root,
        n_trials=2,
        max_depth=15,
        batch_mode="combinatorial",
        sampler_seed=7,
        population_size=4,
    )

    assert wave["trial_count"] == 2
    assert wave["cell_count"] == 12
    assert wave["scientific_contract"]["beam"]["enabled"] is True
    assert wave["scientific_contract"]["optuna_variables_only"] == [
        "macro_phase1_cap",
        "macro_phase2_cap",
        "child_phase1_cap",
        "child_phase2_cap",
        "batch_search_pool_size",
        "batch_size_cap",
        "lambda_add",
    ]
    for trial_ref in wave["trials"]:
        trial_manifest = json.loads(
            (wave_root / trial_ref["trial_manifest"]).read_text(encoding="utf-8")
        )
        assert trial_manifest["regime_order"] == list(route_optuna.REGIME_ORDER)
        assert len(trial_manifest["cells"]) == 6
        assert trial_manifest["policy"]["lambda_add"] in {0.1, 0.75}
        assert "macro_phase1_cap" in trial_manifest["policy"]
        assert "macro_phase2_cap" in trial_manifest["policy"]
        assert "child_phase1_cap" in trial_manifest["policy"]
        assert "child_phase2_cap" in trial_manifest["policy"]
        assert len({cell["regime"] for cell in trial_manifest["cells"]}) == 6
    study = optuna.load_study(
        study_name="route-a-test",
        storage=route_optuna._storage_url(database),
    )
    assert sum(trial.state == optuna.trial.TrialState.RUNNING for trial in study.trials) == 2

            "all_branch_query_work_status_diagnostic": "ok",

                "depth": 20.0 + index,
            },
            "controller_round_count": 15,
            "ansatz_depth": 20 + index,
            "nfev_total": 200 + index,
            "joint_selector_summary": {"selected_cardinality": 2},
        },
    }


def test_aggregate_wave_uses_worst_error_arithmetic_mean_and_resource_sums(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    database = tmp_path / "study.sqlite3"
    wave_root = tmp_path / "wave"
    result_root = wave_root / "results"
    aggregate_root = tmp_path / "aggregate"
    wave = route_optuna.plan_wave(
        database=database,
        study_name="aggregate-test",
        output_dir=wave_root,
        n_trials=1,
        max_depth=15,
        batch_mode="combinatorial",
        sampler_seed=11,
        population_size=4,
    )
    monkeypatch.setattr(
        route_optuna,
        "validate_locked_pareto_plan",
        lambda *args, **kwargs: {"status": "pass"},
    )
    trial_manifest_path = wave_root / wave["trials"][0]["trial_manifest"]
    trial_manifest = json.loads(trial_manifest_path.read_text(encoding="utf-8"))
    for index, cell_ref in enumerate(trial_manifest["cells"]):
        cell_manifest = json.loads(
            (wave_root / cell_ref["manifest"]).read_text(encoding="utf-8")
        )
        output_dir = result_root / cell_manifest["output_relative_dir"]
        _write_json(
            output_dir / "result.json",
            _fake_result(str(cell_ref["regime"]), index),
        )

    summary = route_optuna.aggregate_wave(
        database=database,
        study_name="aggregate-test",
        wave_manifest=wave_root / "wave_manifest.json",
        result_root=result_root,
        output_dir=aggregate_root,
    )

    assert summary["status"] == "complete"
    assert summary["told_complete"] == 1
    aggregate_path = Path(summary["trial_aggregates"][0]["aggregate_json"])
    aggregate = json.loads(aggregate_path.read_text(encoding="utf-8"))
    assert aggregate["objectives"] == {
        "worst_regime_abs_delta_e": pytest.approx(0.06),
        "total_expanded_query_work": pytest.approx(
            sum(1_000.0 + i for i in range(6))
        ),
        "total_graph_count_2q_proxy": pytest.approx(sum(10.0 + i for i in range(6))),
        "total_graph_depth_proxy": pytest.approx(sum(20.0 + i for i in range(6))),
    }
    assert aggregate["query_axis_contract"] == "all_expanded_scored_branches_v1"
    assert aggregate["secondary_diagnostics"]["arithmetic_mean_abs_delta_e"] == pytest.approx(0.035)
    assert aggregate["secondary_diagnostics"][
        "total_winning_branch_query_work"
    ] == pytest.approx(sum(100.0 + i for i in range(6)))
    study = optuna.load_study(
        study_name="aggregate-test",
        storage=route_optuna._storage_url(database),
    )
    complete = [
        trial for trial in study.trials if trial.state == optuna.trial.TrialState.COMPLETE
    ]
    assert len(complete) == 1
    assert complete[0].values == pytest.approx(
        [
            0.06,
            sum(1_000.0 + i for i in range(6)),
            sum(10.0 + i for i in range(6)),
            sum(20.0 + i for i in range(6)),
        ]
    )
    assert all(
        row["graph_cost_proxy_provenance"]["posthoc_fallback_used"] is False
        for row in aggregate["regime_results"]
    )

def test_wave11_legal_chtc_export_transfers_and_preflights_runtime_provenance(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    fake_repo = tmp_path / "repo"
    fake_repo.mkdir()
    checkpoint_rel = Path("raw_outputs/source/wave11/current.json")
    legal_plan_rel = Path("raw_outputs/source/wave18/plan.json")
    checkpoint = fake_repo / checkpoint_rel
    legal_plan = fake_repo / legal_plan_rel
    checkpoint.parent.mkdir(parents=True)
    legal_plan.parent.mkdir(parents=True)
    checkpoint.write_text('{"checkpoint": true}\n', encoding="utf-8")
    legal_plan.write_text('{"legal_plan": true}\n', encoding="utf-8")
    wrapper = fake_repo / "wrapper.sh"
    wrapper.write_text("#!/usr/bin/env bash\n", encoding="utf-8")
    source = fake_repo / "source.py"
    source.write_text("VALUE = 1\n", encoding="utf-8")
    monkeypatch.setattr(route_optuna, "REPO_ROOT", fake_repo)
    monkeypatch.setattr(route_optuna, "WAVE11_SOURCE_CHECKPOINT", checkpoint_rel)
    monkeypatch.setattr(route_optuna, "WAVE11_LEGAL_SOURCE_PLAN", legal_plan_rel)
    monkeypatch.setattr(route_optuna, "CHTC_WRAPPER", Path("wrapper.sh"))
    monkeypatch.setattr(
        route_optuna,
        "CHTC_SOURCE_HASH_PATHS",
        ("source.py", "wrapper.sh"),
    )
    wave_root = fake_repo / "wave"
    route_optuna.plan_wave11_legal_matrix(output_dir=wave_root)
    export_dir = fake_repo / "chtc" / "input" / "wave11_legal"

    export = route_optuna.export_chtc_wave(
        wave_manifest=wave_root / "wave_manifest.json",
        output_dir=export_dir,
        batch_output_root=Path("raw_outputs/wave11_legal_results"),
        batch_name="wave11-legal-test",
    )
    preflight = route_optuna.validate_chtc_export(
        export_dir / "export_manifest.json"
    )

    assert preflight["status"] == "pass"
    assert export["runtime_provenance_inputs"] == [
        {
            "role": "legal_projection_reference_plan",
            "path": str(legal_plan_rel),
            "sha256": route_optuna._sha256_file(legal_plan),
        },
        {
            "role": "wave11_unchecked_checkpoint",
            "path": str(checkpoint_rel),
            "sha256": route_optuna._sha256_file(checkpoint),
        },
    ]
    transfer_inputs = route_optuna._submit_transfer_inputs(
        (export_dir / "submit.sub").read_text(encoding="utf-8")
    )
    assert str(checkpoint_rel) in transfer_inputs
    assert str(legal_plan_rel) in transfer_inputs

    missing_manifest_entry = dict(export)
    missing_manifest_entry.pop("runtime_provenance_inputs")
    route_optuna._write_json(
        export_dir / "export_manifest.json",
        route_optuna._manifest_with_hash(missing_manifest_entry),
    )
    with pytest.raises(ValueError, match="runtime provenance manifest"):
        route_optuna.validate_chtc_export(export_dir / "export_manifest.json")

    submit_path = export_dir / "submit.sub"
    submit_text = submit_path.read_text(encoding="utf-8")
    submit_path.write_text(
        submit_text.replace(f", {checkpoint_rel}", ""),
        encoding="utf-8",
    )
    missing_transfer = dict(export)
    missing_transfer["submit_sha256"] = route_optuna._sha256_file(submit_path)
    route_optuna._write_json(
        export_dir / "export_manifest.json",
        route_optuna._manifest_with_hash(missing_transfer),
    )
    with pytest.raises(ValueError, match='"worker_visible": false'):
        route_optuna.validate_chtc_export(export_dir / "export_manifest.json")
