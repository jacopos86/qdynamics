def _write_json(path: Path, payload: dict) -> None:
    path.parent.mkdir(parents=True, exist_ok=True)
    path.write_text(json.dumps(payload, indent=2), encoding="utf-8")


def _digest(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()

def _alignment(tmp_path: Path, *, strong_weak_u: float = 8.0) -> Path:
    rows = []
    display_labels = {
        "strong-weak-u8": "strong-weak",
        "strong-strong-u8": "strong-strong",
    }
    for regime in EXPECTED_REGIMES:
        payload = _result_payload(regime)
        if regime == "strong-weak-u8":
            payload["settings"]["u"] = strong_weak_u
        result_json = tmp_path / regime / "json" / "result.json"
        _write_json(result_json, payload)
        rows.append(
            {
                "method": "SNAKE",
                "regime": display_labels.get(regime, regime),
                "source_json": str(result_json),
                "source_sha256": _digest(result_json),
                "display": {"k_pl": 13, "DeltaE": "4.71e-4"},
                "one_minus_F_display": "0.000216",
            }
        )
    alignment = tmp_path / "alignment.json"
    _write_json(
        alignment,
        {
            "schema": "unit",
            "snake_rows": rows,
        },
    )
    return alignment


def test_full_policy_warm_start_manifest_extracts_six_visible_rows(tmp_path: Path) -> None:
    payload = build_warm_start_manifest(_alignment(tmp_path))

    assert payload["schema"] == "paper_i_hh_snake_full_policy_warm_start_manifest_v1"
    assert tuple(payload["regimes"]) == EXPECTED_REGIMES
    weak = payload["regimes"]["weak-weak"]["enqueue_params"][0]["params"]
    assert weak["base_preset"] == "resolved_default"
    assert weak["runtime_split_mode"] == "shortlist_pauli_children_v1"
    assert weak["selection_cost_mode"] == "marrakesh_graph_span_v1"
    assert weak["full_phase2_batch_target_size"] == "8"
    assert weak["full_phase1_prune_fraction"] == "0.4101910583864897"
    assert weak["full_spsa_maxiter"] == "200"
    assert "200" in payload["search_space"]["full_spsa_maxiter"]
    assert payload["fixed_identity_locks"]["prune_policy"] == "recoverability_ladder_v1"


def test_full_policy_warm_start_rejects_ambiguous_legacy_strong_label(tmp_path: Path) -> None:
    with pytest.raises(ValueError, match="Ambiguous or unsupported visible regime"):
        build_warm_start_manifest(_alignment(tmp_path, strong_weak_u=1.25))
