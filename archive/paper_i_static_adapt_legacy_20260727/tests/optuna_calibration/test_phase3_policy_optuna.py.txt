def _spec(problem: str = "hubbard") -> HamiltonianBenchmarkSpec:
    return HamiltonianBenchmarkSpec(
        benchmark_id=f"{problem}_L2",
        family=problem,
        features=ProblemFeatureVector(
            problem=problem,
            size_label="L2",
            L=2,
            n_qubits=4,
            pool_size_hint=100,
            spinful=True,
        ),
        base_pipeline_args=("--problem", problem, "--L", "2", "--adapt-max-depth", "12"),
        baseline_abs_delta_e=1e-3,
        baseline_count_2q=100,
        baseline_depth_2q=300,
        baseline_parameter_count=20,
    )


def test_paper_i_artifacts_preserve_payload_exact_audit_non_hit() -> None:
    classification = {
        "schema_version": "static_adapt_target_hit_classification_v1",
        "source": "adapt_vqe",
        "target_hit_success": False,
        "status": "noisy_oracle_target_hit_exact_audit_non_hit",
        "non_hit_reason": "exact_final_state_audit_misses_target",
        "terminal_stop_reason": "benchmark_abs_delta_e_target",
        "required_stop_reason": "benchmark_abs_delta_e_target",
        "target_configured": True,
        "target_error": 2.1e-14,
        "target_threshold": 2e-4,
        "target_error_within_threshold": True,
        "exact_final_state_audit_required": True,
        "exact_final_state_target_error": 0.05,
        "exact_final_state_target_error_within_threshold": False,
    }
    payload = {
        "adapt_vqe": {
            "benchmark_target_classification": classification,
            "continuation": {"benchmark_target_classification": classification},
            "history": [],
        }
    }
    result = BenchmarkResult(
        benchmark_id="hubbard_L2",
        family="hubbard",
        success=True,
        abs_delta_e=2e-14,
        abs_delta_e_reference=2e-14,
        stop_reason="benchmark_abs_delta_e_target",
    )
    physical_target = {
        "schema": "paper_i_physical_target_v1",
        "tau_phys": 2e-4,
        "tau_tight": 2e-4,
        "accuracy_gate_metric": "abs_error_reference_cutoff",
        "legacy_same_cutoff_thresholds": [],
    }

    _, _, first_crossing = p3opt._paper_i_artifacts_for_result(
        result,
        _spec(),
        result_payload=payload,
        physical_target=physical_target,
    )

    assert first_crossing["target_hit_success"] is False
    assert first_crossing["target_hit_classification"]["target_hit_success"] is False
    assert first_crossing["target_hit_classification"]["non_hit_reason"] == "exact_final_state_audit_misses_target"


def test_size_scaled_budget_grows_with_pool_and_qubits_but_caps() -> None:
    features = ProblemFeatureVector(problem="extended_hubbard", size_label="L4", L=4, n_qubits=8, pool_size_hint=500)
    budget = SizeScaledBudget(min_count=16, max_count=128, pool_fraction=0.5, qubit_slope=4.0)
    assert budget.resolve(features) == 128

    small = replace(features, n_qubits=4, pool_size_hint=10)
    assert budget.resolve(small) == 16

    ("kwargs", "match"),

        ({"hardware_resolution_mode": "manual"}, "one of"),
    ),
)

    kwargs: dict[str, object],
    match: str,
) -> None:

    with pytest.raises(ValueError, match=match):
        policy_to_cli_args(policy, _spec("hh"))

def test_cost_at_accuracy_objective_stops_rewarding_extra_accuracy_and_penalizes_shots() -> None:
    spec = replace(_spec(), baseline_shot_cost_proxy=10.0)
    weights = StaticObjectiveWeights(
        energy=1.0,
        count_2q=0.0,
        circuit_depth=0.0,
        parameters=0.0,
        shot_cost=1.0,
        target_abs_delta_e=1e-5,
    )
    accurate_low_shot = BenchmarkResult(
        benchmark_id=spec.benchmark_id,
        family=spec.family,
        success=True,
        abs_delta_e=1e-10,
        shot_cost_proxy=10.0,
        measurement_proxy_validated=True,
    )
    accurate_high_shot = BenchmarkResult(
        benchmark_id=spec.benchmark_id,
        family=spec.family,
        success=True,
        abs_delta_e=1e-14,
        shot_cost_proxy=100.0,
        measurement_proxy_validated=True,
    )
    inaccurate = BenchmarkResult(
        benchmark_id=spec.benchmark_id,
        family=spec.family,
        success=True,
        abs_delta_e=1e-3,
        shot_cost_proxy=10.0,
        measurement_proxy_validated=True,
    )
    inaccurate_tiny_shot = BenchmarkResult(
        benchmark_id=spec.benchmark_id,
        family=spec.family,
        success=True,
        abs_delta_e=1e-2,
        count_2q=1,
        circuit_depth=1,
        parameter_count=1,
        shot_cost_proxy=0.0,
        measurement_proxy_validated=True,
    )

    assert normalized_static_score(accurate_low_shot, spec, weights) == pytest.approx(0.0)
    assert normalized_static_score(accurate_high_shot, spec, weights) > normalized_static_score(accurate_low_shot, spec, weights)
    assert normalized_static_score(inaccurate, spec, weights) > normalized_static_score(accurate_low_shot, spec, weights)
    assert normalized_static_score(inaccurate_tiny_shot, spec, weights) > normalized_static_score(accurate_high_shot, spec, weights)

    invalid_proxy_high_shot = replace(accurate_high_shot, measurement_proxy_validated=False)
    assert normalized_static_score(invalid_proxy_high_shot, spec, weights) == pytest.approx(
        normalized_static_score(accurate_low_shot, spec, weights)
    )

def test_objective_global_agnostic_can_parallelize_benchmarks_with_one_policy(tmp_path: Path) -> None:
    import threading
    import time

    specs = (_spec("hubbard"), _spec("extended_hubbard"), _spec("ionic_hubbard"))
    lock = threading.Lock()
    active = 0
    max_active = 0
    policies = []

    def _runner(spec, policy, *, output_dir):
        nonlocal active, max_active
        with lock:
            active += 1
            max_active = max(max_active, active)
            policies.append(policy)
        try:
            time.sleep(0.05)
            return BenchmarkResult(
                benchmark_id=spec.benchmark_id,
                family=spec.family,
                success=True,
                abs_delta_e=5e-4,
                count_2q=120,
                circuit_depth=330,
                parameter_count=24,
            )
        finally:
            with lock:
                active -= 1

    trial = _FakeTrial()
    value = objective_global_agnostic(
        trial,
        specs,
        output_dir=tmp_path,
        runner=_runner,
        benchmarks_per_trial_jobs=0,
    )

    assert value < 0.0
    assert max_active > 1
    assert len(policies) == len(specs)
    assert len({id(policy) for policy in policies}) == 1
    assert trial.user_attrs["benchmarks_per_trial_jobs"] == 0
    assert set(trial.user_attrs["results"]) == {spec.benchmark_id for spec in specs}

def test_run_subprocess_logged_kills_registered_process_group_on_exception(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    class _FakeProcess:
        pid = 43210

        def __init__(self) -> None:
            self.wait_calls = 0

        def wait(self, timeout=None):
            self.wait_calls += 1
            if self.wait_calls == 1:
                raise RuntimeError("wait exploded")
            return -int(signal.SIGTERM)

    fake_proc = _FakeProcess()
    kill_calls: list[tuple[int, signal.Signals]] = []

    def _fake_popen(*_args, **_kwargs):
        return fake_proc

    def _fake_killpg(pid: int, sig: signal.Signals) -> None:
        kill_calls.append((int(pid), sig))

    monkeypatch.setattr(p3opt.subprocess, "Popen", _fake_popen)
    monkeypatch.setattr(p3opt.os, "killpg", _fake_killpg)

    lifecycle = p3opt._ManagedOptunaRunLifecycle(run_id="unit-test")
    with pytest.raises(RuntimeError, match="wait exploded"):
        p3opt._run_subprocess_logged(
            [sys.executable, "-c", "pass"],
            cwd=tmp_path,
            stdout_path=tmp_path / "stdout.log",
            stderr_path=tmp_path / "stderr.log",
            run_lifecycle=lifecycle,
            subprocess_label="adapt",
            benchmark_id="hubbard_L2",
            trial_number=0,
        )

    assert kill_calls == [(43210, signal.SIGTERM)]
    assert lifecycle.active_process_count() == 0

def test_main_auto_robustness_gate_runs_before_broad_global_study(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    order: list[str] = []
    specs = (_spec("hubbard"), _spec("extended_hubbard"))

    monkeypatch.setattr(p3opt, "filter_static_benchmark_suite", lambda **_kwargs: specs)

    def fake_gate(**kwargs):
        order.append("gate")
        assert kwargs["output_dir"] == tmp_path
        assert kwargs["lanes"] == ("SPSA", "POWELL")
        assert kwargs["target_abs_delta_e"] == pytest.approx(3e-5)
        return {"schema": "phase3_hubbard_l2_robustness_gate_v1", "ok": True}

    def fake_run_optuna_study(**kwargs):
        order.append("study")
        assert kwargs["specs"] == specs
        return {"ok": True}

    monkeypatch.setattr(p3opt, "_run_phase3_robustness_gate_preflight", fake_gate)
    monkeypatch.setattr(p3opt, "run_optuna_study", fake_run_optuna_study)

    rc = p3opt.main(
        [
            "--mode",
            "global",
            "--output-dir",
            str(tmp_path),
            "--n-trials",
            "1",
            "--robustness-gate",
            "auto",
            "--robustness-gate-target-abs-delta-e",
            "3e-5",
            "--no-enqueue-historical",
        ]
    )

    assert rc == 0
    assert order == ["gate", "study"]


def test_main_robustness_gate_failure_stops_before_study(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
    capsys: pytest.CaptureFixture[str],
) -> None:
    specs = (_spec("hubbard"), _spec("extended_hubbard"))
    monkeypatch.setattr(p3opt, "filter_static_benchmark_suite", lambda **_kwargs: specs)
    monkeypatch.setattr(
        p3opt,
        "_run_phase3_robustness_gate_preflight",
        lambda **_kwargs: {"schema": "phase3_hubbard_l2_robustness_gate_v1", "ok": False},
    )

    def fail_if_called(**_kwargs):  # pragma: no cover - assertion guard
        raise AssertionError("run_optuna_study should not run after a failed robustness gate")

    monkeypatch.setattr(p3opt, "run_optuna_study", fail_if_called)

    rc = p3opt.main(
        [
            "--mode",
            "global",
            "--output-dir",
            str(tmp_path),
            "--n-trials",
            "1",
            "--robustness-gate",
            "require",
            "--no-enqueue-historical",
        ]
    )

    assert rc == 2
    assert "robustness gate failed" in capsys.readouterr().err.lower()
