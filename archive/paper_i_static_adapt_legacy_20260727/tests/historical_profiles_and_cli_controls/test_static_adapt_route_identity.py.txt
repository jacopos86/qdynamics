def test_static_route_identity_config_helper_matches_direct_route_payload() -> None:

    assert observed["base_pool_key"] == "full_meta"

    assert observed["phase3_enable_batching"] is True

    expected = route_identity_payload(
        observed,

    assert actual == expected
    assert actual["valid"] is True

def test_static_route_identity_base_pool_key_preserves_hh_staged_fallback() -> None:
    assert (
        resolve_route_identity_base_pool_key(
            pool_key="phase3_v1",
            pool_key_input=None,
            hh_staged_continuation=True,
        )
        == "full_meta"
    )
    assert (
        resolve_route_identity_base_pool_key(
            pool_key="phase3_v1",
            pool_key_input="math_md_full_meta",
            hh_staged_continuation=True,
        )
        == "math_md_full_meta"
    )
    assert (
        resolve_route_identity_base_pool_key(
            pool_key="phase3_v1",
            pool_key_input=None,
            hh_staged_continuation=False,
        )
        == "phase3_v1"
    )

    observed = build_static_route_identity_observed_components(
        _static_route_identity_config(
            pool_key="phase3_v1",
            pool_key_input=None,
            hh_staged_continuation=True,
        )
    )
    assert observed["base_pool_key"] == "full_meta"

def test_static_route_identity_helper_validation_raises_declared_mismatch() -> None:
    cfg = _static_route_identity_config(
        phase3_selector_policy="hardware_resolvable_v1",
        algebraic_shortlisting_enabled=False,
    )

    with pytest.raises(ValueError, match="Declared static_route_id=route_a does not match") as exc:

        "mismatch:phase3_selector_policy:'hardware_resolvable_v1'!='algebraic_nested_v1'"
        in str(exc.value)
    )

        _current_route_a_components(
            phase3_plateau_acquisition_mode="off",
            phase3_plateau_acquisition_score="log_volume_v1",
            phase3_plateau_score_formula="log(1 + sigma_perp_lambda / lambda_vol) / (1 + K3)",
            phase3_plateau_duplicate_policy="block_exact_position_v1",
        ),
        declared_route_id=ROUTE_ID_C,
    )

    assert payload["valid"] is False
    assert "mismatch:phase3_plateau_acquisition_mode:'off'!='novelty_cost_v1'" in payload[
        "noncanonical_reasons"
    ]


def test_route_c_preserves_route_b_legacy_pairwise_distinction() -> None:
    payload = route_identity_payload(
        _route_c_components(
            phase2_novelty_mode="legacy_pairwise_v1",
            phase2_raw_score_formula="DeltaE_TR_raw * N2_pairwise / (1 + K2)",
        ),
        declared_route_id=ROUTE_ID_C,
    )

    assert payload["valid"] is False
    assert any(str(reason).startswith("mismatch:phase2_novelty_mode") for reason in payload["noncanonical_reasons"])


def test_normalize_static_route_id_accepts_route_c_aliases() -> None:
    assert route_identity_payload(_route_c_components(), declared_route_id="route-c")["route_id"] == ROUTE_ID_C
    assert route_identity_payload(_route_c_components(), declared_route_id="plateau_acquisition_v1")["route_id"] == ROUTE_ID_C


def test_static_route_id_from_record_detects_route_c_plateau_mode() -> None:
    assert static_route_id_from_record({"phase3_plateau_acquisition_mode": "novelty_cost_v1"}) == ROUTE_ID_C
    with pytest.raises(ValueError, match="Route-C-named"):
        static_route_id_from_record(
            {"record_id": "routeC_plateau_foundation"},
            record_id="routeC_plateau_foundation",
            fail_on_route_named_missing=True,
        )


def test_static_route_id_from_record_fails_closed_on_mixed_route_b_and_c_telemetry() -> None:
    mixed = {
        "phase2_novelty_mode": "legacy_pairwise_v1",
        "phase3_plateau_acquisition_mode": "novelty_cost_v1",
    }

    assert static_route_id_from_record(mixed) == ROUTE_ID_UNSPECIFIED
    with pytest.raises(ValueError, match="mixes legacy Route-B"):
        static_route_id_from_record(mixed, record_id="mixed", fail_on_route_named_missing=True)


def test_route_b_rejects_collective_span_raw_formula() -> None:
    payload = route_identity_payload(
        _current_route_a_components(phase2_novelty_mode="legacy_pairwise_v1"),
        declared_route_id=ROUTE_ID_B_LEGACY_PAIRWISE,
    )

    assert payload["valid"] is False
    assert any(str(reason).startswith("mismatch:phase2_raw_score_formula") for reason in payload["noncanonical_reasons"])


def test_route_named_records_must_declare_route_id() -> None:
    with pytest.raises(ValueError, match="Route-A-named"):
        static_route_id_from_record(
            {"record_id": "routeA_spsa_mixed"},
            record_id="routeA_spsa_mixed",
            fail_on_route_named_missing=True,
        )
    assert (
        static_route_id_from_record(
            {"static_route_id": "route_a"},
            record_id="routeA_spsa_mixed",
            fail_on_route_named_missing=True,
        )
        == ROUTE_ID_A
    )
    assert static_route_id_from_record({}) == ROUTE_ID_UNSPECIFIED
