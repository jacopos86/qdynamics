def _sha256(path: Path) -> str:
    return hashlib.sha256(path.read_bytes()).hexdigest()


def _result_payload(regime: str, *, energy: float = -1.0) -> dict[str, object]:
    exact = -1.1
    return {
        "regime": regime,
        "adapt_vqe": {
            "success": True,
            "stop_reason": "max_depth",
            "history": [{"outer_iteration": 1}, {"outer_iteration": 2}],
            "ansatz_depth": 2,
            "operators": [
                {"label": "first", "terms": [{"pauli_exyz": "xy"}]},
                {"label": "second", "terms": [{"pauli_exyz": "yx"}]},
            ],
            "energy": energy,
            "exact_gs_energy": exact,
            "abs_delta_e": abs(energy - exact),
        },
    }


def _source_argv(
    result_json: Path, *, u: str, estimator_ledger_json: Path | None = None
) -> list[str]:
    return [
        "python3",
        "-m",
        "pipelines.static_adapt.adapt_pipeline",
        "--problem",
        "hh",
        "--adapt-continuation-mode",
        "phase3_v1",
        "--static-route-id",
        "route_a",
        "--static-lane-route",
        "physical_operator_type",
        "--phase2-no-batching",
        "--phase3-no-batching",
        "--phase3-runtime-split-mode",
        "shortlist_pauli_children_v1",
        "--phase3-runtime-split-selection-mode",
        "archival_child_set_forward_v1",
        "--phase3-runtime-split-max-subset-size",
        "1",
        "--u",
        u,
        "--adapt-estimator-call-ledger-json",
        str(
            estimator_ledger_json
            if estimator_ledger_json is not None
            else result_json.with_name("estimator_call_ledger.json")
        ),
        "--output-json",
        str(result_json),
    ]


def _write_source_commands(tmp_path: Path) -> tuple[Path, dict[str, Path]]:
    results: dict[str, Path] = {}
    rows = []
    for regime, u in (("weak-weak", "0.25"), ("strong-weak", "8.0")):
        result = tmp_path / regime.replace("-", "_") / "json" / "result.json"
        result.parent.mkdir(parents=True)
        result.write_text(
            json.dumps(_result_payload(regime)) + "\n",
            encoding="utf-8",
        )
        estimator_ledger = result.with_name("estimator_call_ledger.json")
        estimator_ledger.write_text("{}\n", encoding="utf-8")
        argv = _source_argv(
            result, u=u, estimator_ledger_json=estimator_ledger
        )
        rows.append(
            {
                "regime": regime,
                "argv": argv,
                "shell": shlex.join(argv),
                "output_json": str(result),
            }
        )
        results[regime] = result
    source = tmp_path / "commands.json"
    source.write_text(json.dumps(rows, indent=2) + "\n", encoding="utf-8")
    return source, results


def _options(argv: list[str]) -> dict[str, str | None]:
    _prefix, values = campaign._parse_argv(argv)
    return values

def test_prelaunch_verification_and_parent_preparation_are_no_clobber(
    tmp_path: Path,
) -> None:
    source, _results = _write_source_commands(tmp_path)
    campaign_root = tmp_path / "planned"
    payload = campaign.build_campaign(
        source_commands_json=source,
        regimes=["weak-weak"],
        campaign_root=campaign_root,
    )
    record = payload["records"][0]

    receipt = campaign.verify_campaign_manifest_prelaunch(
        payload,
        record_id=record["record_id"],
        current_cwd=campaign.REPO_ROOT,
    )
    assert receipt["status"] == "pass"
    assert receipt["execution_authorized"] is False
    assert receipt["target_paths_absent"] is True
    assert set(receipt["verified_target_paths"]) == {
        *campaign.RECORD_TARGET_PATH_FIELDS,
        *(f"cache:{name}" for name in campaign.RECORD_CACHE_ENVIRONMENT_VARIABLES),
    }

    prepared = campaign.prepare_record_parent_directories(
        record, campaign_root=campaign_root
    )
    assert prepared == record["required_parent_directories"]
    assert all(Path(path).is_dir() for path in prepared)
    assert not Path(record["log_path"]).exists()

    Path(record["log_path"]).write_text("occupied\n", encoding="utf-8")
    with pytest.raises(ValueError, match="no-clobber"):
        campaign.verify_campaign_manifest_prelaunch(
            payload,
            record_id=record["record_id"],
            current_cwd=campaign.REPO_ROOT,
        )
    with pytest.raises(ValueError, match="no-clobber"):
        campaign.prepare_record_parent_directories(
            record, campaign_root=campaign_root
        )

def test_current_revision_control_validation_binds_pair_without_historical_claim(
    tmp_path: Path,
) -> None:
    source, _results = _write_source_commands(tmp_path)
    payload = campaign.build_campaign(
        source_commands_json=source,
        regimes=["weak-weak"],
        campaign_root=tmp_path / "planned-current",
    )
    record = next(
        row for row in payload["records"] if row["profile_key"] == "disabled"
    )
    output = Path(record["output_json"])
    output.parent.mkdir(parents=True, exist_ok=True)
    # This intentionally differs from the historical source result.  The
    # current-revision receipt binds the fresh control without claiming that
    # it reproduced the historical anchor.
    completed = _result_payload("weak-weak", energy=-0.97)
    completed["settings"] = {
        "sr_escape_mode": "disabled",
        "historical_singleton_coordinate_solve_policy": record[
            "historical_singleton_coordinate_solve_policy"
        ],
        "historical_singleton_trust_region_update_policy": record[
            "historical_singleton_trust_region_update_policy"
        ],
        "sr_powell_coordinate_chart_policy": record[
            "sr_powell_coordinate_chart_policy"
        ],
    }
    output.write_text(json.dumps(completed) + "\n", encoding="utf-8")

    receipt = campaign.validate_current_revision_disabled_result(
        payload, record_id=record["record_id"]
    )
    assert receipt["status"] == "pass"
    assert receipt["schema"] == (
        campaign.CURRENT_REVISION_CONTROL_VALIDATION_SCHEMA
    )
    assert receipt["historical_equivalence_claimed"] is False
    assert receipt["execution_authorized"] is False
    assert receipt["combined_mode_authorized"] is False
    assert receipt["paired_saddle_output_absent"] is True
    assert all(receipt["result_profile_checks"].values())

    completed["settings"]["sr_escape_mode"] = "saddle_only"
    output.write_text(json.dumps(completed) + "\n", encoding="utf-8")
    with pytest.raises(ValueError, match="control profile mismatch"):
        campaign.validate_current_revision_disabled_result(
            payload, record_id=record["record_id"]
        )


def test_current_revision_control_validation_rejects_ambiguous_or_cross_regime_pair(
    tmp_path: Path,
) -> None:
    source, _results = _write_source_commands(tmp_path)
    payload = campaign.build_campaign(
        source_commands_json=source,
        regimes=["weak-weak"],
        campaign_root=tmp_path / "planned-current-pair",
    )
    record = next(
        row for row in payload["records"] if row["profile_key"] == "disabled"
    )
    output = Path(record["output_json"])
    output.parent.mkdir(parents=True, exist_ok=True)
    completed = _result_payload("weak-weak")
    completed["settings"] = {
        "sr_escape_mode": "disabled",
        "historical_singleton_coordinate_solve_policy": record[
            "historical_singleton_coordinate_solve_policy"
        ],
        "historical_singleton_trust_region_update_policy": record[
            "historical_singleton_trust_region_update_policy"
        ],
        "sr_powell_coordinate_chart_policy": record[
            "sr_powell_coordinate_chart_policy"
        ],

            cross_regime, record_id=record["record_id"]

def test_source_row_integrity_fails_closed_before_generation(tmp_path: Path) -> None:

            regimes=["weak-weak"],
            campaign_root=tmp_path / "planned",
            repo_root=tmp_path,
        )


def test_cli_writes_only_the_manifest_and_reports_fail_closed(
    tmp_path: Path, capsys: pytest.CaptureFixture[str]
) -> None:
    source, results = _write_source_commands(tmp_path)
    manifest = tmp_path / "campaign.json"

    assert (
        campaign.main(
            [
                "--source-commands-json",
                str(source),
                "--regime",
                "strong_weak",
                "--campaign-root",
                str(tmp_path / "planned"),
                "--output-manifest",
                str(manifest),
            ]
        )
        == 0
    )
    written = json.loads(manifest.read_text(encoding="utf-8"))
    assert written["launch_count"] == 0
    assert written["requested_regimes"] == ["strong-weak"]
    assert (
        campaign.main(
            [
                "--verify-manifest",
                str(manifest),
                "--record-id",
                "strong_weak__sr_disabled",
            ]
        )
        == 0
    )

    results["strong-weak"].unlink()
    blocked_manifest = tmp_path / "blocked.json"
    assert (
        campaign.main(
            [
                "--source-commands-json",
                str(source),
                "--regime",
                "strong-weak",
                "--campaign-root",
                str(tmp_path / "planned-blocked"),
                "--output-manifest",
                str(blocked_manifest),
            ]
        )
        == 2
    )
    assert not blocked_manifest.exists()
    assert "FAIL CLOSED" in capsys.readouterr().err
