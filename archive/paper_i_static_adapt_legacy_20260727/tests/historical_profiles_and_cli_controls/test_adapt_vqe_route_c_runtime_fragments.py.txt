    def test_route_c_plateau_successful_unlock_commits_trial_and_promotes_nonzero_coords(self):
        payload, _ = _run_hardcoded_adapt_vqe(
            **self._route_c_plateau_kwargs(
                adapt_drop_floor=1e9,
                adapt_drop_patience=1,
                adapt_drop_min_depth=1,
                phase3_plateau_unlock_margin=1e-3,
                adapt_final_full_refit=False,
            )
        )

        route_c = payload["route_c_plateau_acquisition"]
        assert route_c["enabled"] is True
        assert int(route_c["unlock_count"]) >= 1
        assert any(row.get("event") == "successful_unlock" for row in route_c["events_tail"])
        assert route_c["exit_status"] in {"unlocked", "dormant_records_remaining"}

        success_rows = [
            row
            for row in payload["history"]
            if isinstance(row.get("route_c_plateau_acquisition"), dict)
            and row["route_c_plateau_acquisition"].get("event") == "successful_unlock"
        ]
        assert success_rows
        success = success_rows[-1]["route_c_plateau_acquisition"]
        assert success["unlock_success"] is True
        assert success["unlock_status"] == "successful_unlock"
        assert float(success["trial_drop"]) + 1e-14 >= float(success["unlock_margin"])
        assert success_rows[-1]["energy_after_opt"] == pytest.approx(float(success["trial_energy"]))
        assert success_rows[-1]["energy_after_opt"] == pytest.approx(
            float(success["reportable_energy_after"])
        )
        assert float(success["reportable_energy_after"]) < float(success["reportable_energy_before"])

        promoted = [int(i) for i in success["promoted_logical_indices"]]
        remaining = [int(i) for i in success["remaining_dormant_indices"]]
        success_state = success["state_after"]
        assert promoted
        assert set(promoted).isdisjoint(set(remaining))
        assert bool(success_state["active_episode"]) is bool(remaining)
        assert int(success_state["unlock_count"]) >= 1
        theta_zero = float(route_c["theta_zero"])
        assert success["logical_index_space"] == "route_c_event_post_insertion_pre_final_prune"
        event_parameter_count = int(success["logical_parameter_count_after_trial"])
        promoted_abs_theta = {
            int(k): float(v) for k, v in success["promoted_logical_abs_theta"].items()
        }
        for idx in promoted:
            assert 0 <= int(idx) < event_parameter_count
            assert promoted_abs_theta[int(idx)] > theta_zero
        logical_theta = [float(x) for x in payload["logical_optimal_point"]]
        for idx in remaining:
            assert abs(logical_theta[idx]) == pytest.approx(0.0, abs=1e-14)
        assert sorted(remaining) == sorted(int(i) for i in route_c["dormant_logical_indices"])
        assert int(route_c["dormant_count"]) == len(remaining)
    def test_route_c_plateau_second_failed_unlock_trains_existing_dormant_coordinate(self):
        payload, _ = _run_hardcoded_adapt_vqe(
            **self._route_c_plateau_kwargs(
                max_depth=3,
                adapt_drop_floor=1e9,
                adapt_drop_patience=1,
                adapt_drop_min_depth=1,
                phase3_plateau_unlock_margin=1e9,
            )
        )

        failed_rows = [
            row
            for row in payload["history"]
            if isinstance(row.get("route_c_plateau_acquisition"), dict)
            and row["route_c_plateau_acquisition"].get("event") == "failed_unlock_dormant_admission"
        ]
        assert len(failed_rows) >= 2
        first_failed = failed_rows[0]["route_c_plateau_acquisition"]
        second_row = failed_rows[1]
        second_failed = second_row["route_c_plateau_acquisition"]
        first_dormants = [int(i) for i in first_failed["dormant_indices_after"]]
        assert first_dormants
        assert set(first_dormants).issubset(
            {int(i) for i in second_failed["dormant_indices_before"]}
        )

        selected_record = second_failed["selected_record"]
        assert isinstance(selected_record, dict)
        window = selected_record["active_dormant_nested_window"]
        assert isinstance(window, dict)
        assert set(first_dormants).issubset({int(i) for i in window["dormant_old_pre_indices"]})
        assert selected_record["phase3_plateau_novelty_source"] == "active_dormant_schur_v1"
        assert selected_record["phase3_plateau_novelty_error"] is None
        assert set(first_dormants).issubset(
            {int(i) for i in selected_record["phase3_plateau_novelty_dormant_indices"]}
        )
        assert set(first_dormants).issubset(
            {int(i) for i in selected_record["phase3_plateau_novelty_context_indices"]}
        )
        assert selected_record["phase3_plateau_novelty_dormant_count"] >= len(first_dormants)
        dormant_post = [int(i) for i in window["dormant_old_post_indices"]]
        trainable = {int(i) for i in second_failed["trainable_logical_indices"]}
        assert dormant_post
        assert set(dormant_post).issubset(trainable)
        assert int(window["candidate_post_index"]) in trainable

        assert second_failed["unlock_success"] is False
        assert second_failed["unlock_status"] == "failed_dormant_commit"
        assert second_row["energy_after_opt"] == pytest.approx(
            float(second_failed["reportable_energy_before"])
        )
        assert second_row["energy_after_opt"] == pytest.approx(
            float(second_failed["reportable_energy_after"])
        )
        assert set(dormant_post).issubset({int(i) for i in second_failed["dormant_indices_after"]})

        route_c = payload["route_c_plateau_acquisition"]
        assert int(route_c["failed_unlock_count"]) >= 2
        logical_theta = [float(x) for x in payload["logical_optimal_point"]]
        for dormant_idx in route_c["dormant_logical_indices"]:
            assert abs(logical_theta[int(dormant_idx)]) == pytest.approx(0.0, abs=1e-14)
        assert payload["final_full_refit"]["executed"] is False
        assert payload["prune_summary"]["suppressed_reason"] == "route_c_plateau_dormant_records_present"
