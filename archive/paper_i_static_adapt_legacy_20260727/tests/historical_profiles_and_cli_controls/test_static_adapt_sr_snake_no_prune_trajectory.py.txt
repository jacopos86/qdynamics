def _read_json(path: Path) -> dict[str, Any]:
    payload = json.loads(path.read_text(encoding="utf-8"))
    assert isinstance(payload, dict)
    return payload

    current_path = output_dir / "current.json"
    ledger_path = output_dir / "ledger.json"
    argv = [
        "--problem",
        "hh",
        "--L",
        "2",
        "--ordering",
        "blocked",
        "--boundary",
        "open",
        "--t",
        "1.0",
        "--u",
        "0.5",
        "--dv",
        "0.0",
        "--omega0",
        "1.0",
        "--g-ep",
        "0.2",

        "1",
        "--boson-encoding",
        "binary",
        "--sr-route-profile",
        "sr_snake_no_prune_symmetric_cost_projected_phase3_no_overlap_trust_v1",
        "--adapt-max-depth",
        str(max_depth),
        "--adapt-current-json-every-depth",
        str(checkpoint_every_depth),
        "--adapt-current-json-keep-history-tail",
        str(keep_history_tail),
        "--adapt-current-json",
        str(current_path),
        "--adapt-estimator-call-ledger-json",
        str(ledger_path),

        argv.extend(

    return argv

def test_private_paper_i_replay_round_lock_fails_closed() -> None:
    request = SRRunRequest(
        method=SRMethodPolicy(insertion=AppendOnlyInsertion()),
        execution=SRExecutionPolicy(
            stop=SRStopPolicy(maximum_controller_rounds=2)
        )
    )
    record = {
        "source_controller_round": 0,
        "source_depth": 0,
        "target_controller_round": 2,
        "target_depth": 2,
        "max_new_admissions": 2,
        "terminal_qiskit_sidecar_outer_iteration": 2,
        "exact_round_horizon_required": True,
        "route_profile": ROUTE_PROFILE,
        "route_profile_contract_sha256": ROUTE_DIGEST,
    }
    lock = _validate_paper_i_replay_round_lock(request, record)
    assert lock.requested_new_rounds == 2
    assert lock.target_controller_round == 2

    inconsistent_fields = {
        "source_depth": 1,
        "target_depth": 3,
        "max_new_admissions": 3,
        "terminal_qiskit_sidecar_outer_iteration": 3,
        "exact_round_horizon_required": False,
    }
    for key, value in inconsistent_fields.items():
        with pytest.raises(ValueError, match="Paper-I replay"):
            _validate_paper_i_replay_round_lock(
                request,
                {**record, key: value},
            )
    with pytest.raises(ValueError, match="requested round limit"):
        _validate_paper_i_replay_round_lock(
            replace(
                request,
                execution=SRExecutionPolicy(
                    stop=SRStopPolicy(maximum_controller_rounds=1)
                ),
            ),
            record,
        )
    for key, value in (
        ("route_profile", "historical_profile"),
        ("route_profile_contract_sha256", "0" * 64),
    ):
        with pytest.raises(ValueError, match="profile or digest"):
            _validate_paper_i_replay_round_lock(
                request,
                {**record, key: value},
            )
    cli_record = {
        "record_id": "weak-weak-r0-r2",
        "target_controller_round": 2,
        "target_depth": 2,
        "max_new_admissions": 2,
        "route_profile": ROUTE_PROFILE,
        "route_profile_contract_sha256": ROUTE_DIGEST,
    }
    cli_lock = _validate_normalized_cli_segment_round_lock(
        request,
        cli_record,
    )
    assert cli_lock.requested_rounds == 2
    for key, value in (
        ("target_controller_round", 3),
        ("target_depth", 3),
        ("max_new_admissions", 3),
    ):
        with pytest.raises(ValueError, match="CLI segment"):
            _validate_normalized_cli_segment_round_lock(
                request,
                {**cli_record, key: value},
            )
    for validator, source, field in (
        (
            _validate_paper_i_replay_round_lock,
            record,
            "target_controller_round",
        ),
        (
            _validate_normalized_cli_segment_round_lock,
            cli_record,
            "target_controller_round",
        ),
    ):
        with pytest.raises(ValueError, match="must be an integer"):
            validator(request, {**source, field: 2.5})

def test_named_legacy_compatibility_dispatch_preserves_reason(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    state = object()
    log_records: list[tuple[str, dict[str, Any]]] = []

    def _fake_legacy_executor(**kwargs: Any) -> tuple[dict[str, Any], object]:
        assert kwargs == {"sentinel": 7}
        return {"success": True}, state

    def _capture_log(event_name: str, **kwargs: Any) -> None:
        log_records.append((event_name, dict(kwargs)))

    monkeypatch.setattr(
        adapt_pipeline,
        "_run_hardcoded_adapt_vqe",
        _fake_legacy_executor,
    )
    monkeypatch.setattr(adapt_pipeline, "_ai_log", _capture_log)
    payload, observed_state = (
        adapt_pipeline._run_hardcoded_adapt_vqe_compatibility(
            compatibility_reason=(
                "normalized_adapt_drop_floor_requires_legacy_compatibility"
            ),
            sentinel=7,
        )
    )
    expected_receipt = {
        "schema": "sr_snake_legacy_compatibility_dispatch_v1",
        "reason": (
            "normalized_adapt_drop_floor_requires_legacy_compatibility"
        ),
        "executor": "_run_hardcoded_adapt_vqe",
        "no_credentials_serialized": True,
    }
    assert observed_state is state
    assert payload["compatibility_dispatch"] == expected_receipt
    assert log_records == [
        (
            "sr_snake_legacy_compatibility_dispatch",
            expected_receipt,
        )
    ]


def test_round_one_checkpoint_survives_forced_postpublication_interruption(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    monkeypatch.setenv("STATIC_ADAPT_CANDIDATE_RECORD_CACHE", "off")
    monkeypatch.setenv("STATIC_ADAPT_HH_POOL_CACHE", "off")
    monkeypatch.chdir(tmp_path)
    output_dir = Path("interrupted_after_round_one")
    output_dir.mkdir()
    original_public_run = sr_runner.run_sr_snake
    completed_results: list[Any] = []

    def _interrupt_after_publication(*args: Any, **kwargs: Any) -> Any:
        result = original_public_run(*args, **kwargs)
        completed_results.append(result)
        raise RuntimeError("issue13_forced_after_round_one_checkpoint")

    monkeypatch.setattr(
        sr_runner,
        "run_sr_snake",
        _interrupt_after_publication,
    )
    argv = _small_route_argv(
        output_dir,
        checkpoint_every_depth=1,
        keep_history_tail=1,
        max_depth=1,
    )
    argv.extend(["--initial-state-source", "hf"])
    with pytest.raises(
        ValueError,
        match="Output SR-SNAKE route contract disagrees",
    ):
        adapt_pipeline.main(argv)

    assert len(completed_results) == 1
    current_path = output_dir / "current.json"
    current = _read_json(current_path)
    adapt = current["adapt_vqe"]
    checkpoint = adapt["terminal_active_prefix_checkpoint"]
    assert adapt["history_count"] == 1
    assert adapt["history_tail_count"] == 1
    assert len(adapt["history"]) == 1
    assert adapt["history_checkpoint_complete"] is True
    assert adapt["history_tail_retention"] == {
        "schema": "static_adapt_verified_resume_history_retention_v1",

        "requested_window_count": 1,
        "serialized_complete_history_count": 1,
        "normalized_for_verified_singleton_resume": True,
        "no_credentials_serialized": True,
    }
    assert adapt["operators"] == EXPECTED_OPERATORS[:1]
    assert checkpoint["outer_iteration"] == 1
    assert checkpoint["active_ansatz_depth"] == 1
    assert (
        checkpoint["checkpoint_sha256"]
        == "01bcaa44acbd6a01e360673fa76229fe8918fe3c17e3ffe90a18eb8f26d08635"
    )
    assert (
        adapt["history"][0]["active_prefix_checkpoint"]["checkpoint_sha256"]
        == "a60bdf2845dfde9a40325da13b8652265ce56951ddc91ad01ab33f7ba2537e53"
    )
    assert checkpoint["checkpoint_kind"] == (
        "terminal_post_final_refit_and_prune"
    )

    resume_pointer = adapt["verified_singleton_resume_sidecar"]
    resume_sidecar = _read_json(
        current_path.with_name(resume_pointer["path"])
    )
    recorded_source_path = Path(resume_sidecar["source_result_json"])
    assert recorded_source_path.is_absolute()
    assert recorded_source_path.resolve() == current_path.resolve()
    resume_source = load_static_resume_source(current_path)
    resumed_ledger = load_checkpoint_estimator_call_ledger(resume_source)
    assert resumed_ledger is not None
    ledger_payload, provenance = resumed_ledger
    assert provenance["restored_prefix_S_alg"] == 299
    assert provenance["restored_prefix_S_unique"] == 250
    assert ledger_payload["occurrence_summary"][
        "total_call_occurrences"
    ] == 299
    assert ledger_payload["summary"]["S_unique"] == 250
    verified_resume = extract_verified_singleton_resume_checkpoint(
        resume_source
    )
    assert verified_resume.controller_round == 1
    assert verified_resume.ansatz_depth == 1
    assert verified_resume.operator_labels == tuple(EXPECTED_OPERATORS[:1])
    assert verified_resume.selected_parent_pool_indices == (0,)
    assert verified_resume.estimator_call_ledger_provenance[
        "restored_prefix_S_alg"
    ] == 299
    assert verified_resume.estimator_call_ledger_provenance[
        "restored_prefix_S_unique"
    ] == 250
    pointer = adapt["estimator_call_ledger_checkpoint"]
    sidecar_path = current_path.with_name(pointer["path"])
    sidecar_bytes = sidecar_path.read_bytes()
    assert hashlib.sha256(sidecar_bytes).hexdigest() == pointer["sha256"]
    current_artifact = next(
        artifact
        for artifact in completed_results[0].observation.artifacts
        if artifact.kind == "accepted_state_checkpoint"
    )
    current_bytes = current_path.read_bytes()
    assert current_artifact.sha256 == hashlib.sha256(current_bytes).hexdigest()
    assert current_artifact.size_bytes == len(current_bytes)

    def _legacy_executor_forbidden(*_args: Any, **_kwargs: Any) -> None:

    monkeypatch.setattr(
        adapt_pipeline,
        "_run_hardcoded_adapt_vqe",
        _legacy_executor_forbidden,
    )
    monkeypatch.setattr(
        selected_qiskit,

    )
    monkeypatch.setattr(
        recovery_qiskit,
        "compile_table_i_pauli_label_groups",
        _legacy_executor_forbidden,
    )
    monkeypatch.setattr(
        recovery_qiskit,
        "compile_circuit_for_backend",
        _legacy_executor_forbidden,
    )
    normalized_route_pairs: list[tuple[str, str]] = []
    route_receipt_requests: list[str] = []
    original_compatibility_reason = (
        adapt_pipeline._active_cli_compatibility_reason
    )
    original_active_cli_run = adapt_pipeline._run_active_cli_sr_snake
    original_public_run = sr_runner.run_sr_snake
    public_results: list[Any] = []
    adapter_projection_keys: list[set[str]] = []

    def _compatibility_reason_spy(
        executor_kwargs: dict[str, Any],
    ) -> str | None:
        normalized_route_pairs.append(
            (
                str(executor_kwargs["sr_route_profile_request"]),

    def _selection_spy(
        state: sr_selection._SRControllerState,
        workspace: sr_selection._SelectionWorkspace,
    ) -> sr_selection._SingletonAdmissionDecision:
        selection_rounds.append(state.controller_round)

        class _KernelSpy:
            def accepted_state_snapshot(self) -> object:
                return workspace.kernel.accepted_state_snapshot()

            def evaluate(

        )

    def _transition_spy(
        state: sr_transition._AcceptedStateSnapshot,
        decision: sr_selection._SingletonAdmissionDecision,
        workspace: sr_transition._TransitionWorkspace,
    ) -> sr_transition._AcceptedSingletonTransition:
        nonlocal transition_active
        transition_rounds.append(state.controller_round)

        class _KernelSpy:
            def execute(
                self,
                exact_decision: sr_selection._SingletonAdmissionDecision,
                live_record: object,
                runtime: sr_transition._TransitionNumericalRuntime,
            ) -> sr_transition._TransitionEvaluation:
                transition_callback_rounds.append(state.controller_round)

                    runtime.accepted_state_snapshot()

                    exact_decision,
                    live_record,
                    runtime,

                state,
                decision,
                replace(workspace, kernel=_KernelSpy()),
            )
        finally:
            transition_active = False
        transition_results.append(result)
        return result
