        grouped_violation_indices={4},

def test_formal_selector_prefilter_rejects_scoring_only_parent_templates() -> None:
    """An execution-unsafe parent may be scored, but it cannot be admitted."""

    records = [
        {"candidate_label": "safe-child"},
        {"candidate_label": "unsafe-parent-template"},
    ]

    def _assert_safe(record) -> None:
        if record["candidate_label"] == "unsafe-parent-template":
            raise RuntimeError("execution representation violates sector")

    retained, rejected = adapt_pipeline._filter_sector_safe_admission_records(
        records,
        assert_safe=_assert_safe,
    )

    assert [row["candidate_label"] for row in retained] == ["safe-child"]
    assert rejected == ["unsafe-parent-template"]

) -> None:

    monkeypatch.setattr(adapt_pipeline, "_ai_log", lambda *args, **kwargs: None)

        def __init__(self, *args, **kwargs):
            raise AssertionError("FM-SNAKE invoked Joint-Response selection logic")

        _ForbiddenJointResponseEvaluator,
    )

        raise AssertionError(
            "pure FM truncated its population before the formal selector"

        _forbidden_generic_tie_beam_prefilter,

        workspace = kwargs["workspace"]
        formal_population_sizes.append(len(workspace.candidate_records))

        adapt_pipeline,
        "select_combinatorial_query_closed_batch",
        _record_formal_population_size,
    )
    current_json = tmp_path / "pure_fm_beam_current.json"

    payload, _psi = adapt_pipeline._run_hardcoded_adapt_vqe(
        h_poly=_two_site_hubbard_hamiltonian(),
        num_sites=2,
        ordering="blocked",
        problem="hubbard",
        adapt_pool="uccsd",
        t=1.0,
        u=4.0,
        dv=0.0,
        boundary="periodic",
        omega0=0.0,
        g_ep=0.0,
        n_ph_max=1,
        boson_encoding="binary",
        max_depth=2,
        eps_grad=1.0e-12,
        eps_energy=0.0,
        maxiter=3,
        seed=7,
        allow_repeats=False,
        finite_angle_fallback=False,
        finite_angle=0.1,
        finite_angle_min_improvement=1.0e-12,
        adapt_state_backend="compiled",
        adapt_reopt_policy="full",
        phase1_prune_enabled=False,
        phase2_enable_batching=False,
        phase3_runtime_split_mode="shortlist_pauli_children_v1",
        phase3_runtime_split_selection_mode="global_child_only_v1",
        phase3_runtime_split_subset_sizes=(1,),
        phase3_runtime_split_child_set_symmetry_policy="hard_guard",
        # A one-record Phase-III carrier must not truncate the authoritative
        # Phase-II population supplied to the formal selector.
        route_a_funnel_config=replace(
            _formal_beam_carrier_config(),
            child_phase3_cap=1,
        ),
        adapt_beam_live_branches=3,
        adapt_beam_children_per_parent=2,
        adapt_beam_terminated_keep=0,
        adapt_reoptimization_route=FORMAL_MANIFOLD_ROUTE,
        adapt_final_full_refit=False,
        adapt_current_json=current_json,
        adapt_current_json_every_depth=1,
        adapt_estimator_call_ledger_enabled=True,
    )

    assert payload["success"] is True
    assert payload["adapt_reoptimization_route"] == FORMAL_MANIFOLD_ROUTE
    assert payload["adapt_beam_enabled"] is True
    assert payload["adapt_beam_live_branches"] == 3
    assert formal_population_sizes
    assert max(formal_population_sizes) > 1
    beam = payload["continuation"]["beam_search"]
    assert beam["beam_enabled"] is True
    assert beam["rounds"]
    assert int(payload["nfev_total"]) == int(beam["winning_branch_nfev"]) + int(
        beam["discarded_branch_nfev_operational_overhead"]
    )
    assert int(payload["nfev_winning_lineage"]) == int(
        beam["winning_branch_nfev"]
    )
    assert int(payload["nfev_discarded_operational_overhead"]) == int(
        beam["discarded_branch_nfev_operational_overhead"]
    )

    winner_id = int(beam["winner_branch_id"])
    winner_branch = f"beam_branch:{winner_id}"
    formal_summary = payload["formal_manifold_warm_start"]
    assert formal_summary["branch_id"] == winner_branch
    assert all(
        row["formal_manifold_warm_start"]["route"] == FORMAL_MANIFOLD_ROUTE
        and str(row["formal_manifold_warm_start"]["branch_id"]).startswith(
            "beam_branch:"
        )
        for row in payload["history"]
    )

    query_closure = payload["formal_manifold_query_closure"]
    assert query_closure["joint_response_selector_invoked"] is False
    winning_ids = set(query_closure["winning_branch"]["unique_primitive_ids"])
    discarded_ids = set(
        query_closure["discarded_branch_operational_overhead"][
            "unique_primitive_ids"
        ]
    )
    assert winning_ids
    assert winning_ids.isdisjoint(discarded_ids)
    all_ids = set(query_closure["all_executed"]["unique_primitive_ids"])
    assert winning_ids | discarded_ids == all_ids
    route_energy_occurrences = query_closure[
        "energy_evaluation_occurrence_ids"
    ]
    assert set(query_closure["energy_evaluation_occurrence_ids"]) == set(
        query_closure["all_executed"]["primitive_ids_by_category"]["N_E"]
    )
    shared_hamiltonian = payload["estimator_call_accounting"][
        "executed_occurrence_accounting"
    ]["all_hamiltonian_execution"]
    shared_scope_counts = shared_hamiltonian[
        "occurrence_count_by_consumer_scope"
    ]
    assert any(
        str(scope).startswith("energy:formal_manifold_beam_local_reopt:")
        and int(count) > 0
        for scope, count in shared_scope_counts.items()
    )
    shared_formal_occurrences = sum(
        int(count)
        for scope, count in shared_scope_counts.items()
        if str(scope).startswith("energy:formal_manifold_")
        or str(scope) == "energy:initial_state"
    )
    assert shared_formal_occurrences == int(payload["nfev_total"])
    assert len(route_energy_occurrences) == int(payload["nfev_total"]) + int(
        shared_scope_counts.get("outer_state_refresh", 0)
    )

    checkpoint = json.loads(current_json.read_text(encoding="utf-8"))
    adapt_checkpoint = checkpoint["adapt_vqe"]
    checkpoint_branch = f"beam_branch:{int(adapt_checkpoint['branch_id'])}"
    warm_checkpoint = adapt_checkpoint["formal_manifold_warm_state_checkpoint"]
    query_checkpoint = adapt_checkpoint[
        "formal_manifold_query_closure_checkpoint"
    ]
    assert warm_checkpoint["branch_id"] == checkpoint_branch
    assert query_checkpoint["selected_branch_id"] == checkpoint_branch
    checkpoint_winning_ids = set(
        query_checkpoint["winning_branch_ledger"]["kind_by_id"]
    )
    checkpoint_discarded_ids = set(
        query_checkpoint["discarded_branch_operational_ledger"]["kind_by_id"]
    )
    assert checkpoint_winning_ids.isdisjoint(checkpoint_discarded_ids)
    assert len(
        query_checkpoint["energy_evaluation_occurrence_ids_winning"]
    ) + len(query_checkpoint["energy_evaluation_occurrence_ids_discarded"]) >= int(
        adapt_checkpoint["nfev_total"]
    )

def test_pure_fm_filters_one_redundant_phase2_candidate_and_selects_another(
    monkeypatch,
) -> None:
    """A typed rank-gate rejection is local to one pure-FM candidate."""

    monkeypatch.setattr(adapt_pipeline, "_ai_log", lambda *args, **kwargs: None)

    def _forbidden_selector(*args, **kwargs):
        raise AssertionError("pure FM invoked JR or SR selection logic")

    for name in (
        "RouteAJointResponseEvaluator",
        "select_route_a_schur_proposals",
        "evaluate_historical_singleton_phase2_coordinate_models",
        "evaluate_historical_singleton_coordinate_models",
        "historical_singleton_scalar_selector_summary",
    ):
        monkeypatch.setattr(adapt_pipeline, name, _forbidden_selector)

    original_selector = adapt_pipeline.select_combinatorial_query_closed_batch
    audit = {
        "forced_rank_gate_count": 0,
        "feasible_singleton_count": 0,
        "forced_candidate_key": None,
    }

    def _one_rank_gate_then_delegate(**kwargs):
        selection = original_selector(**kwargs)
        workspace = kwargs["workspace"]
        if (
            audit["forced_rank_gate_count"] == 0
            and len(workspace.candidate_records) == 1
            and selection.ordered_results
        ):
            result = selection.ordered_results[0]
            failed = replace(
                result,
                feasible=False,
                reason="candidate_subset_supported_rank_gate_failed",
                joint_step=np.zeros_like(result.joint_step),
                predicted_reduction=0.0,
                score=0.0,

                query_free_derived_fields=(),
                solve_result=None,
            )
            audit["forced_rank_gate_count"] = 1
            audit["forced_candidate_key"] = str(
                workspace.candidate_keys[0]
            )
            return replace(
                selection,
                feasible=False,
                reason="no_feasible_second_order_subset",
                selected=None,
                feasible_subset_count=0,
                ordered_results=(failed,),
            )
        if len(workspace.candidate_records) == 1 and selection.feasible:
            audit["feasible_singleton_count"] += 1
        return selection

    monkeypatch.setattr(
        adapt_pipeline,

        _one_rank_gate_then_delegate,

    payload, _psi = adapt_pipeline._run_hardcoded_adapt_vqe(

        problem="hubbard",

        dv=0.0,
        boundary="periodic",
        omega0=0.0,
        g_ep=0.0,
        n_ph_max=1,
        boson_encoding="binary",
        max_depth=1,
        eps_grad=1.0e-12,
        eps_energy=0.0,
        maxiter=1,
        seed=7,
        allow_repeats=True,
        finite_angle_fallback=False,
        finite_angle=0.1,
        finite_angle_min_improvement=1.0e-12,
        adapt_state_backend="compiled",
        adapt_reopt_policy="full",
        adapt_insertion_mode="always",
        phase1_prune_enabled=False,
        phase2_enable_batching=False,
        phase3_runtime_split_mode="shortlist_pauli_children_v1",
        phase3_runtime_split_selection_mode="global_child_only_v1",
        phase3_runtime_split_subset_sizes=(1,),
        phase3_runtime_split_child_set_symmetry_policy="hard_guard",
        route_a_funnel_config=_formal_beam_carrier_config(),
        adapt_beam_live_branches=3,
        adapt_beam_children_per_parent=2,
        adapt_beam_terminated_keep=0,
        adapt_reoptimization_route=FORMAL_MANIFOLD_ROUTE,
        adapt_final_full_refit=False,
    )

    assert audit["forced_rank_gate_count"] == 1
    assert audit["forced_candidate_key"] is not None
    assert audit["feasible_singleton_count"] >= 1
    assert payload["success"] is True
    assert len(payload["history"]) == 1
    history_payload = json.dumps(payload["history"], sort_keys=True)
    assert "candidate_subset_supported_rank_gate_failed" in history_payload
    assert "authoritative_phase2_ineligible_reason_counts" in history_payload
    assert payload["formal_manifold_query_closure"][
        "joint_response_selector_invoked"
    ] is False
