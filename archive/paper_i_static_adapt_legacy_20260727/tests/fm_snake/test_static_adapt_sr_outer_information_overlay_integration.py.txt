def _small_full_meta_filter(directory: Path) -> Path:
    directory.mkdir(parents=True, exist_ok=True)
    path = directory / "active_small_full_meta.json"
    path.write_text(
        json.dumps(
            {
                "keep_classes": ["uccsd_sing"],
                "classifier_version": HH_FULL_META_CLASSIFIER_VERSION,
                "source_pool": "full_meta",
                "source_problem": "hh",
                "source_num_sites": 2,
                "source_n_ph_max": 2,
            },
            sort_keys=True,
        ),
        encoding="utf-8",
    )
    return path


def _sr_overlay_kwargs(
    directory: Path,
    *,
    route_profile: str | None,
    admissions: int = 2,
) -> tuple[dict[str, Any], Path]:
    directory.mkdir(parents=True, exist_ok=True)
    parser = _build_adapt_arg_parser(adapt_gradient_parity_rtol=1.0e-8)
    profile_args = (
        []
        if route_profile is None
        else [
            "--adapt-formal-manifold-route-profile",
            str(route_profile),
        ]
    )
    args = parser.parse_args(
        [
            "--problem",
            "hh",
            "--L",
            "2",
            "--u",
            "0.5",
            "--g-ep",
            "0.2",
            "--omega0",
            "1.0",
            "--n-ph-max",
            "2",
            "--boundary",
            "open",
            "--skip-pdf",
            "--adapt-max-depth",
            "30",
            "--sr-route-profile",
            SR_ROUTE_PROFILE_NO_PRUNE_SYMMETRIC_COST_ALIAS_V1,
            *profile_args,
        ]
    )
    h_poly = build_hubbard_holstein_hamiltonian(
        dims=2,
        J=1.0,
        U=0.5,
        omega0=1.0,
        g=0.2,
        n_ph_max=2,
        boson_encoding="binary",
        repr_mode="JW",
        indexing="blocked",
        pbc=False,
        include_zero_point=True,
    )
    context = resolve_problem_context(
        ProblemRequest(
            problem_key="hh",
            num_sites=2,
            t=1.0,
            u=0.5,
            dv=0.0,
            omega0=1.0,
            g_ep=0.2,
            n_ph_max=2,
            boson_encoding="binary",
            ordering="blocked",
            boundary="open",
        ),
        hamiltonian=h_poly,
    )
    kwargs = _build_run_hardcoded_adapt_vqe_kwargs(
        args,
        h_poly=h_poly,
        resolved_problem_context=context,
        cli_adapt_continuation_mode="phase3_v1",
        adapt_ref_base_depth=0,
        psi_ref_override=None,
        psi_ref_source=None,
        psi_ref_handoff_state_kind=None,
        exact_gs_override=0.0,
        phase3_oracle_gradient_config=None,
        final_noise_audit_config=None,
    )
    current_json = directory / "current.json"
    kwargs.update(
        {
            "adapt_segment_id": (
                "canonical_sr_baseline"
                if route_profile is None
                else f"sr_outer_overlay:{route_profile}"
            ),
            "adapt_segment_max_new_admissions": int(admissions),
            "adapt_pool_class_filter_json": _small_full_meta_filter(directory),
            "adapt_current_json": current_json,
            "adapt_current_json_every_depth": 1,
            "adapt_estimator_call_ledger_enabled": True,
            "eps_grad": 1.0e-8,
        }
    )
    return kwargs, current_json


def _forbid_fm_jr_and_formal_session(
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    monkeypatch.setenv("STATIC_ADAPT_CANDIDATE_RECORD_CACHE", "off")
    monkeypatch.setenv("STATIC_ADAPT_HH_POOL_CACHE", "off")
    monkeypatch.setattr(adapt_pipeline, "_ai_log", lambda *_a, **_k: None)

    def _forbidden_owner(*_args: Any, **_kwargs: Any) -> Any:
        raise AssertionError(
            "public SR outer-information overlay invoked FM/JR-owned code"
        )

    for name in (
        "RouteAJointResponseEvaluator",
        "run_route_a_child_funnel",
        "select_route_a_schur_proposals",
        "select_combinatorial_query_closed_batch",
    ):
        monkeypatch.setattr(adapt_pipeline, name, _forbidden_owner)
    monkeypatch.setattr(FormalManifoldSession, "__init__", _forbidden_owner)

def test_sr_outer_growth_cache_absence_exact_fallback_is_narrow() -> None:
    feature = _deferred_geometry_expansion_feature()
    summary = feature.phase2_joint_geometry_reuse
    missing_identity = ValueError(
        "authoritative_sr_phase3_candidate_identity_missing"
    )

    assert adapt_pipeline._sr_outer_growth_cache_absence_requires_exact_fallback(
        phase3_summary=summary,
        selected_record={"feature": feature},
        error=missing_identity,
    )
    assert not adapt_pipeline._sr_outer_growth_cache_absence_requires_exact_fallback(
        phase3_summary={**summary, "schema": "historical_singleton_coordinate_model_v1"},
        selected_record={"feature": feature},
        error=missing_identity,
    )
    assert not adapt_pipeline._sr_outer_growth_cache_absence_requires_exact_fallback(
        phase3_summary=summary,
        selected_record={"feature": feature},
        error=ValueError("cached candidate label conflicts with selected winner"),
    )

def test_sr_shadow_overlay_preserves_exact_sr_trajectory_query_ledger_and_public_identity(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    _forbid_fm_jr_and_formal_session(monkeypatch)
    baseline_kwargs, _ = _sr_overlay_kwargs(
        tmp_path / "baseline",
        route_profile=None,
    )
    shadow_kwargs, _ = _sr_overlay_kwargs(
        tmp_path / "shadow",
        route_profile=(
            SR_OUTER_INFORMATION_ROUTE_PROFILE_NO_PRUNE_SYMMETRIC_COST_SHADOW_V1
        ),
    )

    baseline, _ = adapt_pipeline._run_hardcoded_adapt_vqe(**baseline_kwargs)
    shadow, _ = adapt_pipeline._run_hardcoded_adapt_vqe(**shadow_kwargs)

    _assert_source_locked_public_sr_identity(
        shadow,
        route_profile=(
            SR_OUTER_INFORMATION_ROUTE_PROFILE_NO_PRUNE_SYMMETRIC_COST_SHADOW_V1
        ),
    )
    assert len(baseline["history"]) == len(shadow["history"]) == 2
    for baseline_row, shadow_row in zip(
        baseline["history"], shadow["history"], strict=True
    ):
        assert _trajectory_row(shadow_row) == _trajectory_row(baseline_row)
        reuse = shadow_row["sr_outer_information_reuse"]
        assert reuse["mode"] == "shadow_v1"
        assert reuse["route_family"] == "singleton_response_snake"
        assert reuse["fm_selector_invoked"] is False
        assert reuse["jr_selector_invoked"] is False
        assert reuse["structural_rollback_performed"] is False

    assert shadow["energy"] == pytest.approx(
        baseline["energy"], rel=0.0, abs=1.0e-13
    )
    assert shadow["optimal_point"] == pytest.approx(
        baseline["optimal_point"], rel=0.0, abs=1.0e-13
    )
    baseline_ledger = baseline["estimator_call_accounting"]["full_ledger"]
    shadow_ledger = shadow["estimator_call_accounting"]["full_ledger"]
    assert shadow_ledger["ledger_fingerprint"] == baseline_ledger[
        "ledger_fingerprint"
    ]
    assert shadow_ledger["entries"] == baseline_ledger["entries"]
    assert shadow_ledger["occurrences"] == baseline_ledger["occurrences"]
    assert shadow["nfev_total"] == baseline["nfev_total"]

    summary = shadow["formal_manifold_outer_information"]
    assert summary["mode"] == "shadow_v1"
    assert summary["measurement_savings_enabled"] is False
    assert summary["incremental_outer_geometry_query_charge"] == 0
    assert summary["shadow_does_not_claim_sr_measurement_savings"] is True
    assert shadow["measurement_savings_claimed"] is False
    assert shadow["formal_manifold_query_accounting_complete"] is True
    query_checkpoint = shadow["sr_outer_information_query_closure_checkpoint"]
    assert query_checkpoint == shadow[
        "formal_manifold_query_closure_checkpoint"
    ]
    assert query_checkpoint["schema"] == (
        "sr_outer_information_query_checkpoint_v1"
    )
    assert query_checkpoint["current_round_finalized"] is True
    assert query_checkpoint["formal_query_ledger_invoked"] is False
    assert query_checkpoint["fm_selector_invoked"] is False
    assert query_checkpoint["jr_selector_invoked"] is False


def test_sr_active_overlay_reuses_endpoint_hpsi_once_and_checkpoints_nonbeam_prediction(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    _forbid_fm_jr_and_formal_session(monkeypatch)

    original_evaluate = ExactStateBackend.evaluate

    def _forbid_endpoint_full_evaluation(
        self: ExactStateBackend, theta: Any
    ) -> Any:
        if self.manifold_id == "sr_outer_active_phase3_endpoint_v1":
            raise AssertionError(
                "active outer endpoint performed a duplicate full evaluation"
            )
        return original_evaluate(self, theta)

    monkeypatch.setattr(
        ExactStateBackend,
        "evaluate",
        _forbid_endpoint_full_evaluation,
    )

    chart_calls: list[dict[str, Any]] = []
    original_chart = adapt_pipeline.build_supported_fs_powell_chart

    def _trace_chart(*args: Any, **call_kwargs: Any) -> Any:
        receipt = call_kwargs.get("external_logical_fs_gram_receipt")
        chart = original_chart(*args, **call_kwargs)
        chart_calls.append(
            {
                "receipt": receipt,
                "telemetry": dict(chart.base_telemetry),
            }
        )
        return chart

    monkeypatch.setattr(
        adapt_pipeline,
        "build_supported_fs_powell_chart",
        _trace_chart,
    )
    kwargs, current_json = _sr_overlay_kwargs(
        tmp_path / "active",
        route_profile=(
            SR_OUTER_INFORMATION_ROUTE_PROFILE_NO_PRUNE_SYMMETRIC_COST_ACTIVE_V1
        ),
    )

    payload, _ = adapt_pipeline._run_hardcoded_adapt_vqe(**kwargs)

    _assert_source_locked_public_sr_identity(
        payload,
        route_profile=(
            SR_OUTER_INFORMATION_ROUTE_PROFILE_NO_PRUNE_SYMMETRIC_COST_ACTIVE_V1
        ),
    )
    assert len(payload["history"]) == 2
    reuse_rows = [row["sr_outer_information_reuse"] for row in payload["history"]]
    for reuse in reuse_rows:
        assert reuse["mode"] == "active_reuse_v1"
        assert reuse["accepted_reoptimizer_owner"] == (
            SR_OUTER_INFORMATION_ACCEPTED_REOPTIMIZER
        )
        assert reuse["fm_selector_invoked"] is False
        assert reuse["jr_selector_invoked"] is False
        assert reuse["structural_rollback_performed"] is False
        identity = reuse["cached_growth_identity_binding"]
        assert identity["numeric_identity_conflicts_allowed"] is False
        assert identity["numeric_geometry_modified"] is False
        assert identity["authoritative_sr_trajectory_affected"] is False
        growth = reuse["same_ray_growth"]
        assert growth["same_ray_extension_skipped"] is False
        extension = growth["same_ray_extension"]
        assert extension["old_old_metric_reacquired"] is False
        assert growth["candidate_source_primitive_ids"]
        prediction = reuse["refit_prediction"]
        assert prediction["stage"] == "refit_prediction"
        assert prediction["classical_algebra_quantum_query_charge"] == 0

    assert reuse_rows[0]["same_ray_growth"]["same_ray_extension"][
        "old_coordinate_count"
    ] == 0
    assert reuse_rows[1]["same_ray_growth"]["same_ray_extension"][
        "old_coordinate_count"
    ] == 1
    endpoint = reuse_rows[1]["endpoint_preparation"]
    assert endpoint["nfev_delta"] == 0
    gradient_accounting = endpoint["endpoint_gradient_query_accounting"]
    assert gradient_accounting["active_coordinate_count"] == 1
    assert gradient_accounting["new_unique_gradients_charged"] == 1
    assert gradient_accounting["measurement_reused_by_phase3"] is True

    full_ledger = payload["estimator_call_accounting"]["full_ledger"]
    endpoint_scope = "historical_singleton_whitening_active_gradient"
    endpoint_occurrences = [
        occurrence
        for occurrence in full_ledger["occurrences"]
        if occurrence["consumer_scope"] == endpoint_scope
    ]
    assert len(endpoint_occurrences) == 1
    assert endpoint_occurrences[0]["component"] == "N_grad"
    endpoint_entries = [
        entry
        for entry in full_ledger["entries"]
        if any(
            consumer["scope"] == endpoint_scope
            for consumer in entry["consumers"]
        )
    ]
    assert len(endpoint_entries) == 1
    assert endpoint_entries[0]["charged_component"] == "N_grad"
    hamiltonian_scopes = payload["estimator_call_accounting"][
        "executed_occurrence_accounting"
    ]["all_hamiltonian_execution"]["occurrence_count_by_consumer_scope"]
    assert int(
        hamiltonian_scopes.get(
            "energy:formal_manifold_outer_information_closure", 0
        )
    ) == 0

    assert chart_calls
    assert len(chart_calls) >= len(payload["history"])
    reused_chart_calls = [
        row
        for row in chart_calls
        if isinstance(row["receipt"], ExternalLogicalFSGramReceipt)
    ]
    assert len(reused_chart_calls) >= len(payload["history"])
    for chart_call in reused_chart_calls:
        telemetry = chart_call["telemetry"]
        assert telemetry["metric_input_status"] == "reused"
        assert telemetry["metric_backend_evaluation_performed"] is False
        assert telemetry["metric_element_count_acquired_for_chart"] == 0
    for history_row in payload["history"]:
        accepted = history_row["accepted_refit"]
        receipt = accepted["external_logical_fs_gram_receipt"]
        assert receipt is not None
        assert accepted["metric_input_status"] == "reused"

    candidate_ids = {
        str(primitive_id)
        for reuse in reuse_rows
        for primitive_id in reuse["same_ray_growth"][
            "candidate_source_primitive_ids"
        ]
    }
    entry_ids = {str(entry["primitive_id"]) for entry in full_ledger["entries"]}
    assert candidate_ids
    assert candidate_ids.issubset(entry_ids)

    current = json.loads(current_json.read_text(encoding="utf-8"))[
        "adapt_vqe"
    ]
    assert current["formal_manifold_warm_start"] is None
    assert current["formal_manifold_warm_state_checkpoint"] is None
    assert current["formal_manifold_runtime_checkpoint"] is None
    checkpoint = current["sr_outer_information_checkpoint"]
    assert checkpoint == current["formal_manifold_outer_information_checkpoint"]
    assert checkpoint["stage"] == "refit_prediction"
    assert checkpoint["pending"] is not None
    assert checkpoint["pending"]["registry"]
    assert checkpoint["pending"]["theta"]
    assert checkpoint["pending"]["provenance"]
    restored = OuterInformationSession.from_checkpoint_payload(checkpoint)
    assert restored.checkpoint_payload()["checkpoint_sha256"] == checkpoint[
        "checkpoint_sha256"
    ]
    checkpoint_query = current[
        "sr_outer_information_query_closure_checkpoint"
    ]
    assert checkpoint_query["schema"] == (
        "sr_outer_information_query_checkpoint_v1"
    )
    assert checkpoint_query["current_round_finalized"] is True

    accounting = payload["estimator_call_accounting"]
    assert accounting["complete"] is True
    assert accounting["exact_blockers"] == []
    assert accounting["formal_manifold_shared_estimator_bridge"][
        "scalar_occurrences_reconciled"
    ] is True
    assert payload["formal_manifold_query_accounting_complete"] is True
    assert payload["nfev_reconciled"] is True
    assert payload["estimator_energy_occurrence_nfev"] == payload["nfev_total"]
    query = payload["sr_outer_information_query_closure"]
    assert query == payload["formal_manifold_query_closure"]
    assert query["route"] == "singleton_response_snake"
    assert query["primitive_set_reconciliation"] == {
        "winning_discarded_disjoint": True,
        "union_equals_all_executed": True,
        "winning_count": query["all_executed"]["unique_primitive_count"],
        "discarded_count": 0,
        "all_executed_count": query["all_executed"]["unique_primitive_count"],
    }
    assert query["all_executed"]["S_alg"] == sum(
        int(query["all_executed"][name])
        for name in (
            "N_E",
            "N_grad",
            "N_G",
            "N_Q",
            "N_Hv",
            "N_cross",
        )
    )
    assert query["all_executed"]["S_alg"] == full_ledger["summary"]["S_alg"]
    assert query["outer_information_measurement_savings_claimed"] is bool(
        query["outer_information_active_reuse"][
            "old_old_metric_elements_avoided"
        ]
        > 0
    )


def test_sr_active_reuse_does_not_count_planned_phase3_pairs_as_upstream(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    """Only acquired Phase-II pairs may reduce the avoided-old-old count."""

    _forbid_fm_jr_and_formal_session(monkeypatch)
    prior_backed_phase3_contexts: list[dict[str, Any]] = []
    original_prepare_scaffold = (
        adapt_pipeline.Phase2NoveltyOracle.prepare_scaffold_context
    )

    def _trace_scaffold_context(
        self: Any, *args: Any, **call_kwargs: Any
    ) -> Any:
        if call_kwargs.get("old_old_metric_prior") is not None:
            prior_backed_phase3_contexts.append(
                {
                    "metric": np.asarray(
                        call_kwargs["old_old_metric_prior"], dtype=float
                    ),
                    "hessian": np.asarray(
                        call_kwargs["old_old_hessian_prior"], dtype=float
                    ),
                    "status": call_kwargs.get(
                        "old_old_hessian_prior_status"
                    ),
                }
            )
        return original_prepare_scaffold(
            self, *args, **call_kwargs
        )

    monkeypatch.setattr(

        admissions=1,
    )
    kwargs["adapt_parallel_gradient_workers"] = 1
    fresh, _ = adapt_pipeline._run_hardcoded_adapt_vqe(**kwargs)

    with adapt_candidate_record_cache._CANDIDATE_RECORD_CACHE_LOCK:
        adapt_candidate_record_cache._CANDIDATE_RECORD_MEMORY_CACHE.clear()
    cache_sources: list[str] = []
    cache_get = adapt_pipeline._candidate_record_cache_get

    def _trace_cache_get(**call_kwargs: Any) -> tuple[Any, str]:
        record, source = cache_get(**call_kwargs)
        cache_sources.append(str(source))
        return record, source

    monkeypatch.setattr(
        adapt_pipeline,
        "_candidate_record_cache_get",
        _trace_cache_get,
    )
    cached, _ = adapt_pipeline._run_hardcoded_adapt_vqe(**kwargs)

    assert "disk" in cache_sources
    assert [_trajectory_row(row) for row in cached["history"]] == [
        _trajectory_row(row) for row in fresh["history"]
    ]

    cached_ledger = cached["estimator_call_accounting"]["full_ledger"]
    assert cached_ledger["ledger_fingerprint"] == fresh_ledger[
        "ledger_fingerprint"
    ]

    assert cached_ledger["occurrences"] == fresh_ledger["occurrences"]
