    frame = build_supported_frame(
        _metric(),
        config=_config(),
        status=STATUS_ACQUIRED,
        source_primitive_ids=("metric:0",),
    )

    assert frame.rank == 2
    np.testing.assert_allclose(
        frame.raw_orthonormalizer.T

        np.eye(2),
        atol=2.0e-12,
    )
    np.testing.assert_allclose(
        frame.solver_whitening.T

            frame.raw_metric
            + frame.metric_ridge
            * frame.retained_vectors
            @ frame.retained_vectors.T
        )
        @ frame.solver_whitening,
        np.eye(2),
        atol=2.0e-12,
    )


def test_same_ray_extension_inherits_old_block_and_drops_redundant_candidate() -> None:
    old = build_supported_frame(
        np.asarray([[2.0, 0.2], [0.2, 1.0]]),
        config=_config(),
        status=STATUS_ACQUIRED,
        source_primitive_ids=("old",),
    )
    # This candidate is exactly the first raw tangent direction.
    cross = old.raw_metric[:, [0]]
    self_gram = old.raw_metric[np.ix_([0], [0])]
    result = extend_same_ray_frame(
        old,
        active_candidate_gram=cross,
        candidate_candidate_gram=self_gram,
        old_to_new_registry_mapping=(0, 2),
        candidate_positions=(1,),
        config=_config(),
        source_primitive_ids=("candidate-cross", "candidate-self"),
    )

    assert result.frame.rank == old.rank
    assert result.telemetry["residual_supported_rank"] == 0
    assert result.telemetry["old_old_metric_reacquired"] is False
    np.testing.assert_allclose(
        result.augmented_metric[np.ix_((0, 2), (0, 2))],
        old.raw_metric,
        atol=1.0e-12,
    )
    np.testing.assert_allclose(result.residual_gram, 0.0, atol=2.0e-12)

def test_same_ray_extension_canonicalizes_a_just_subthreshold_residual() -> None:
    config = _config()
    old = build_supported_frame(
        np.eye(2),
        config=config,
        status=STATUS_ACQUIRED,
        source_primitive_ids=("old",),
    )
    # The augmented raw Gram is positive, but its candidate residual is just
    # below the shared support threshold.  The inherited raw frame is therefore
    # valid only modulo that discarded quotient direction.
    result = extend_same_ray_frame(
        old,
        active_candidate_gram=np.asarray([[1.0], [0.0]]),
        candidate_candidate_gram=np.asarray([[1.0 + 1.0722672220103232e-7]]),
        old_to_new_registry_mapping=(0, 1),
        candidate_positions=(2,),
        config=config,
        source_primitive_ids=("candidate-cross", "candidate-self"),
    )

    assert result.frame.rank == old.rank
    assert result.telemetry["residual_supported_rank"] == 0
    assert result.telemetry["quotient_frame_canonicalization_applied"] is True
    retained_metric = (
        result.frame.retained_vectors
        @ np.diag(result.frame.retained_eigenvalues)
        @ result.frame.retained_vectors.T
    )
    np.testing.assert_allclose(
        result.frame.raw_analysis.T @ result.frame.raw_analysis,
        retained_metric,
        atol=config.support_span_tolerance,
    )


def test_affine_metric_prediction_is_covariant_under_parameter_rescaling() -> None:
    config = _config()
    G_theta = np.asarray([[2.0, 0.3], [0.3, 1.1]])
    scale = np.asarray([[3.0, 0.0], [0.0, 0.25]])
    G_phi = scale.T @ G_theta @ scale
    theta_frame = build_supported_frame(
        G_theta,
        config=config,
        status=STATUS_ACQUIRED,
        source_primitive_ids=("theta",),
    )
    phi_frame = build_supported_frame(
        G_phi,
        config=config,
        status=STATUS_ACQUIRED,
        source_primitive_ids=("phi",),
    )
    # Q maps phi-frame coefficients into the theta physical frame.
    Q = theta_frame.raw_analysis @ scale @ phi_frame.raw_orthonormalizer
    np.testing.assert_allclose(Q.T @ Q, np.eye(2), atol=2.0e-12)
    H_theta = np.asarray([[0.12, -0.03], [-0.03, -0.08]])
    H_phi = Q.T @ H_theta @ Q
    predicted_theta = predict_endpoint_metric(
        theta_frame,
        normalized_strain=H_theta,
        step_scale=0.7,
        config=config,
    )
    predicted_phi = predict_endpoint_metric(
        phi_frame,
        normalized_strain=H_phi,
        step_scale=0.7,
        config=config,
    )

    np.testing.assert_allclose(
        predicted_phi.frame.raw_metric,
        scale.T @ predicted_theta.frame.raw_metric @ scale,
        atol=2.0e-11,
        rtol=2.0e-11,
    )
    z_theta = np.asarray([0.2, -0.4])
    z_phi = Q.T @ z_theta
    physical_theta = theta_frame.raw_orthonormalizer @ z_theta
    physical_from_phi = scale @ phi_frame.raw_orthonormalizer @ z_phi
    np.testing.assert_allclose(physical_from_phi, physical_theta, atol=2.0e-12)


def test_predicted_support_threshold_crossing_requires_exact_reanchor() -> None:
    config = OuterInformationConfig(
        rank_relative_tolerance=1.0e-6,
        metric_regularization=1.0e-9,
    )
    source = build_supported_frame(
        np.diag([1.0, 1.2e-6]),
        config=config,
        status=STATUS_ACQUIRED,
        source_primitive_ids=("near-threshold-anchor",),
    )
    assert source.rank == 2

    with pytest.raises(OuterInformationReanchorRequired) as caught:
        predict_endpoint_metric(
            source,
            # Supported coordinates are ordered by the retained eigensystem;
            # amplify the unit-eigenvalue mode so the near-threshold mode falls
            # below the shared relative-support test.
            normalized_strain=np.diag([0.0, 8.0]),
            step_scale=1.0,
            config=config,
        )

    assert caught.value.reason == (
        "predicted_endpoint_metric_support_rank_changed"
    )
    telemetry = dict(caught.value.telemetry)
    assert telemetry["source_rank"] == 2
    assert telemetry["predicted_rank"] == 1
    assert telemetry["next_action"] == (
        "discard_prediction_and_require_exact_reanchor"
    )
    assert telemetry["structural_rollback_performed"] is False
    assert telemetry["classical_quantum_query_charge"] == 0


def test_transport_preserves_pairings_and_typed_curvature_actions() -> None:
    config = _config()
    source = build_supported_frame(
        np.asarray([[1.8, 0.15], [0.15, 0.9]]),
        config=config,
        status=STATUS_ACQUIRED,
        source_primitive_ids=("g0",),
    )
    prediction = predict_endpoint_metric(
        source,
        normalized_strain=np.asarray([[0.1, 0.02], [0.02, -0.05]]),
        step_scale=0.6,
        config=config,
    )
    omega = np.asarray([[0.0, 0.2], [-0.2, 0.0]])
    transport = build_split_connection_transport(
        source,
        prediction.frame,
        config=config,
        skew_connection=omega,
        step_scale=0.6,
    )
    vector = np.asarray([0.3, -0.7])
    covector = np.asarray([1.1, 0.4])
    direct = np.asarray([[2.0, 0.2], [0.2, 1.3]])
    inverse = np.linalg.inv(direct)
    vector_new = transport_vector(transport, vector)
    covector_new = transport_covector(transport, covector)
    direct_new = transport_direct_curvature(transport, direct)
    inverse_new = transport_inverse_curvature(transport, inverse)

    assert covector_new @ vector_new == pytest.approx(covector @ vector, abs=1e-12)
    np.testing.assert_allclose(
        direct_new @ vector_new,
        transport_covector(transport, direct @ vector),
        atol=2.0e-12,
    )
    np.testing.assert_allclose(inverse_new, np.linalg.inv(direct_new), atol=2e-12)

def test_closure_pass_correct_and_anchor_are_distinct() -> None:
    config = _config()
    predicted = predict_endpoint_metric(
        build_supported_frame(
            np.asarray([[1.7, 0.1], [0.1, 0.8]]),
            config=config,
            status=STATUS_ACQUIRED,
            source_primitive_ids=("g",),
        ),
        normalized_strain=np.zeros((2, 2)),
        step_scale=1.0,
        config=config,
    ).frame
    X = np.eye(2)

    passed = close_predicted_geometry(
        predicted,
        probe_coefficients=X,
        observed_probe_gram=np.diag([1.0 + 5e-5, 1.0 - 5e-5]),
        config=config,
        source_primitive_ids=("probe-pass",),
    )
    corrected = close_predicted_geometry(
        predicted,
        probe_coefficients=X,
        observed_probe_gram=np.diag([1.01, 0.99]),
        config=config,
        source_primitive_ids=("probe-correct",),
    )
    anchored = close_predicted_geometry(
        predicted,
        probe_coefficients=X,
        observed_probe_gram=np.diag([1.2, 0.8]),
        config=config,
        source_primitive_ids=("probe-anchor",),
    )

    assert passed.outcome == CLOSURE_CONTINUE
    assert passed.frame is predicted
    assert corrected.outcome == CLOSURE_CORRECT
    assert corrected.frame is not None
    assert corrected.frame.status == STATUS_CORRECTED
    np.testing.assert_allclose(
        corrected.frame.raw_orthonormalizer.T
        @ corrected.frame.raw_metric
        @ corrected.frame.raw_orthonormalizer,
        np.eye(2),
        atol=2.0e-12,
    )
    assert anchored.outcome == CLOSURE_ANCHOR
    assert anchored.frame is None

@pytest.mark.parametrize(
    "curvature_kind",
    [CURVATURE_DIRECT, CURVATURE_INVERSE],
)
def test_refit_gradient_prediction_includes_typed_curvature_displacement(
    curvature_kind: str,
) -> None:
    session = OuterInformationSession(
        config=_config(
            endpoint_gradient_prediction_policy=(
                "transported_source_plus_curvature_displacement_v2"
            )
        ),
        committed=_initial_state(curvature_kind=curvature_kind),
    )
    session.begin_same_ray_growth(
        registry=("a", "b", "c"),
        theta=(0.2, -0.1, 0.0),
        active_candidate_gram=np.asarray([[0.08], [0.04]]),
        candidate_candidate_gram=np.asarray([[0.7]]),
        candidate_gradient=(0.12,),
        old_to_new_registry_mapping=(0, 1),
        candidate_positions=(2,),
        source_primitive_ids=("candidate-cross", "candidate-self"),
    )
    source = session.pending
    assert source is not None
    endpoint = np.asarray([0.24, -0.05, 0.06])
    transport = session.predict_refit(
        physical_state_fingerprint="ray:curvature-gradient",
        theta=endpoint,
    )
    predicted = session.pending
    assert predicted is not None
    transported_gradient = transport_covector(
        transport, source.gradient_covector
    )
    displacement_source = source.frame.raw_analysis @ (
        endpoint - source.theta
    )
    displacement_target = transport_vector(
        transport, displacement_source
    )
    if curvature_kind == CURVATURE_DIRECT:
        transported_curvature = transport_direct_curvature(
            transport, source.curvature
        )
        action = transported_curvature @ displacement_target
    else:
        transported_curvature = transport_inverse_curvature(
            transport, source.curvature
        )
        action = np.linalg.solve(
            transported_curvature, displacement_target
        )
    np.testing.assert_allclose(
        predicted.gradient_covector,
        transported_gradient + action,
        atol=2.0e-12,
    )
    curvature_records = [
        record
        for record in predicted.provenance
        if record.quantity == "predicted_endpoint_curvature_transport"
    ]
    assert curvature_records[-1].certificate[
        "endpoint_gradient_prediction_policy"
    ] == "transported_source_plus_curvature_displacement_v2"

def test_transaction_rollback_restores_frame_curvature_and_keeps_measurement_ids() -> None:
    committed = _initial_state()

    )
    assert result.outcome == CLOSURE_ANCHOR
    assert session.stage == TRANSACTION_REANCHOR_REQUIRED

        session.checkpoint_payload()
    )

    assert restored.failed_closure is not None
    assert restored.failed_closure.outcome == CLOSURE_ANCHOR
    assert restored.failed_closure.reason == result.reason

        restored.failed_closure.observed_probe_gram,

        key: value

        if key != "checkpoint_sha256"
    }

        OuterInformationSession.from_checkpoint_payload(forged_failure)

    assert restored.stage == TRANSACTION_IDLE
    assert restored.failed_closure is None
    assert restored.committed.state_id == committed.state_id
    assert restored.pending is None
    assert restored.committed.frame.frame_id == committed.frame.frame_id
    np.testing.assert_allclose(restored.committed.curvature, committed.curvature)
    assert "failed-probe" in restored.immutable_primitive_ids
    assert "failed-probe" not in restored.immutable_observation_ids
    assert "failed-cache-observation" in restored.immutable_observation_ids

def test_checkpoint_resume_preserves_pending_transport_and_rejects_tampering() -> None:
    session = OuterInformationSession(config=_config(), committed=_initial_state())
    _stage_prediction(session)
    payload = session.checkpoint_payload()
    restored = OuterInformationSession.from_checkpoint_payload(payload)

    assert restored.stage == session.stage
    assert restored.committed.state_id == session.committed.state_id
    assert restored.pending.state_id == session.pending.state_id
    np.testing.assert_allclose(
        restored.pending.transport_from_parent,
        session.pending.transport_from_parent,
    )
    assert restored.pending.frame.whitening_provenance_id == (
        session.pending.frame.whitening_provenance_id
    )

    tampered = deepcopy(payload)
    tampered["pending"]["frame"]["frame_id"] = "forged"
    with pytest.raises(ValueError, match="fingerprint mismatch"):
        OuterInformationSession.from_checkpoint_payload(tampered)

    tampered_transport = deepcopy(payload)
    tampered_transport["pending"]["transport_from_parent"][0][0] += 0.125
    with pytest.raises(ValueError, match="parent transport|fingerprint mismatch"):
        OuterInformationSession.from_checkpoint_payload(tampered_transport)

    tampered_config = deepcopy(payload)
    tampered_config["config"]["closure_pass_tolerance"] = 0.049
    with pytest.raises(ValueError, match="checkpoint fingerprint mismatch"):
        OuterInformationSession.from_checkpoint_payload(tampered_config)

    tampered_whitening = deepcopy(payload)
    tampered_whitening["pending"]["frame"]["solver_whitening"][0][0] += 5.0
    unsigned = {
        key: value
        for key, value in tampered_whitening.items()
        if key != "checkpoint_sha256"
    }
    tampered_whitening["checkpoint_sha256"] = outer_module._json_hash(unsigned)
    with pytest.raises(ValueError, match="shared whitening|fingerprint mismatch"):
        OuterInformationSession.from_checkpoint_payload(tampered_whitening)

    forged_stage = deepcopy(payload)
    forged_stage["stage"] = "closure_passed"
    forged_stage["pending_source"] = None
    unsigned = {
        key: value
        for key, value in forged_stage.items()
        if key != "checkpoint_sha256"
    }
    forged_stage["checkpoint_sha256"] = outer_module._json_hash(unsigned)
    with pytest.raises(ValueError, match="transaction mismatch"):
        OuterInformationSession.from_checkpoint_payload(forged_stage)

def test_closure_learns_noncommuting_affine_strain_by_congruence() -> None:
    session = OuterInformationSession(config=_config(), committed=_initial_state())
    _stage_prediction(session)
    predicted = session.pending
    assert predicted is not None
    prior = np.asarray(predicted.normalized_strain_model, dtype=float)
    observed = np.asarray(
        [
            [1.012, 0.008, 0.0],
            [0.008, 0.991, 0.006],
            [0.0, 0.006, 1.004],
        ],
        dtype=float,
    )
    result = session.close_prediction(
        probe_coefficients=np.eye(3),
        observed_probe_gram=observed,
        source_primitive_ids=("noncommuting-endpoint-probe",),
    )
    assert result.outcome == CLOSURE_CORRECT
    half = expm(0.5 * prior)
    composed = 0.5 * (half @ observed @ half + half @ observed.T @ half)
    values, vectors = np.linalg.eigh(composed)
    expected = vectors @ np.diag(np.log(values)) @ vectors.T
    np.testing.assert_allclose(
        session.pending.normalized_strain_model,
        expected,
        atol=2.0e-12,
        rtol=2.0e-12,
    )

def test_cached_geometry_provenance_is_not_misclassified_as_query_primitive() -> None:
    session = OuterInformationSession(config=_config(), committed=_initial_state())
    extension = session.begin_same_ray_growth(
        registry=("a", "b", "candidate"),
        theta=(0.2, -0.1, 0.0),
        active_candidate_gram=np.asarray([[0.1], [0.2]]),
        candidate_candidate_gram=np.asarray([[1.0]]),
        candidate_gradient=(0.3,),
        old_to_new_registry_mapping=(0, 1),
        candidate_positions=(2,),
        source_primitive_ids=(),
        cache_observation_provenance_ids=("cached-phase3-proof",),
        cache_observation_provenance_role=(
            "query_free_cached_sr_v3_phase3_observation"
        ),
        direct_curvature_raw=np.eye(3),
    )

    assert extension.frame.source_primitive_ids == ()
    assert "cached-phase3-proof" not in session.pending.source_primitive_ids
    assert session.pending.source_observation_ids == ("cached-phase3-proof",)
    metric_record = session.pending.provenance[-2]
    curvature_record = session.pending.provenance[-1]
    assert metric_record.primitive_ids == ()
    assert metric_record.observation_ids == ("cached-phase3-proof",)
    assert curvature_record.primitive_ids == ()
    assert curvature_record.observation_ids == ("cached-phase3-proof",)

    restored = OuterInformationSession.from_checkpoint_payload(
        session.checkpoint_payload()
    )
    assert restored.pending.source_observation_ids == ("cached-phase3-proof",)
    assert set(restored.immutable_primitive_ids) == {
        "anchor-gradient",
        "anchor-metric",
    }
    assert "cached-phase3-proof" not in restored.immutable_primitive_ids
    assert "cached-phase3-proof" in restored.immutable_observation_ids

    conflated = deepcopy(session.checkpoint_payload())
    conflated["immutable_observation_ids"].remove("cached-phase3-proof")
    conflated["immutable_primitive_ids"].append("cached-phase3-proof")
    unsigned = {
        key: value
        for key, value in conflated.items()
        if key != "checkpoint_sha256"
    }
    conflated["checkpoint_sha256"] = outer_module._json_hash(unsigned)
    with pytest.raises(ValueError, match="lost immutable observation IDs"):
        OuterInformationSession.from_checkpoint_payload(conflated)

def test_equal_rank_supported_subspace_rotation_has_nonzero_projector_defect() -> None:
    config = _config()
    source = build_supported_frame(
        np.diag([1.0, 0.0]),
        config=config,
        status=STATUS_ACQUIRED,
        source_primitive_ids=("source-metric",),
    )
    angle = 0.1
    rotation = np.asarray(
        [[np.cos(angle), -np.sin(angle)], [np.sin(angle), np.cos(angle)]]
    )
    target_metric = rotation @ np.diag([1.0, 0.0]) @ rotation.T
    target = build_supported_frame(
        target_metric,
        config=config,
        status=STATUS_ACQUIRED,
        source_primitive_ids=("target-metric",),
    )

    assert source.rank == target.rank == 1
    assert supported_subspace_projector_defect(source, target) == pytest.approx(
        np.sin(angle), abs=1.0e-12, rel=0.0
    )
    assert supported_subspace_projector_defect(
        source, target
    ) > config.support_span_tolerance


def _active_direct_growth_session() -> tuple[
    OuterInformationSession, np.ndarray, np.ndarray, tuple[str, ...]
]:
    """Stage one mixed-provenance active same-ray singleton extension."""

    session = OuterInformationSession(
        config=_config(), committed=_initial_state(curvature_kind=CURVATURE_DIRECT)
    )
    metric_cross = np.asarray([[0.07], [-0.04]], dtype=float)
    curvature_cross = np.asarray([[0.3], [-0.2]], dtype=float)
    primitive_ids = (
        "candidate-gradient",
        "candidate-gram-cross",
        "candidate-gram-self",
        "candidate-hessian-cross",
        "candidate-hessian-self",
    )
    session.begin_same_ray_growth(
        registry=("a", "candidate", "b"),
        theta=(0.2, 0.0, -0.1),
        active_candidate_gram=metric_cross,
        candidate_candidate_gram=np.asarray([[1.1]], dtype=float),
        candidate_gradient=(0.12,),
        old_to_new_registry_mapping=(0, 2),
        candidate_positions=(1,),
        source_primitive_ids=primitive_ids,
        cache_observation_provenance_ids=("winning-phase3-observation",),
        cache_observation_provenance_role=(
            "active_outer_reuse_predicted_old_acquired_candidate_blocks"
        ),
        direct_curvature_active_candidate=curvature_cross,
        direct_curvature_candidate_candidate=np.asarray([[0.8]], dtype=float),
    )
    return session, metric_cross, curvature_cross, primitive_ids


def test_active_same_ray_growth_assembles_inherited_old_old_and_acquired_candidate_blocks() -> None:
    session, metric_cross, curvature_cross, primitive_ids = (
        _active_direct_growth_session()
    )
    old = session.committed
    pending = session.pending
    assert old is not None
    assert pending is not None
    old_prior = active_outer_geometry_prior(old)
    pending_prior = active_outer_geometry_prior(pending)
    old_positions = (0, 2)
    candidate_positions = (1,)

    np.testing.assert_allclose(
        pending_prior.raw_metric[np.ix_(old_positions, old_positions)],
        old_prior.raw_metric,
    )
    np.testing.assert_allclose(
        pending_prior.raw_metric[np.ix_(old_positions, candidate_positions)],
        metric_cross,
    )
    np.testing.assert_allclose(
        pending_prior.raw_metric[np.ix_(candidate_positions, candidate_positions)],
        [[1.1]],
    )
    assert old_prior.raw_direct_curvature is not None
    assert pending_prior.raw_direct_curvature is not None
    np.testing.assert_allclose(
        pending_prior.raw_direct_curvature[
            np.ix_(old_positions, old_positions)
        ],
        old_prior.raw_direct_curvature,
    )
    np.testing.assert_allclose(
        pending_prior.raw_direct_curvature[
            np.ix_(old_positions, candidate_positions)
        ],
        curvature_cross,
    )
    np.testing.assert_allclose(
        pending_prior.raw_direct_curvature[
            np.ix_(candidate_positions, candidate_positions)
        ],
        [[0.8]],
    )
    np.testing.assert_allclose(
        pending_prior.raw_gradient[list(old_positions)], old_prior.raw_gradient
    )
    np.testing.assert_allclose(
        pending_prior.raw_gradient[list(candidate_positions)], [0.12]
    )

    metric_record, curvature_record = pending.provenance[-2:]
    assert metric_record.primitive_ids == tuple(sorted(primitive_ids))
    assert curvature_record.primitive_ids == tuple(sorted(primitive_ids))
    assert metric_record.observation_ids == ("winning-phase3-observation",)
    assert curvature_record.observation_ids == ("winning-phase3-observation",)
    assert curvature_record.certificate["old_old_curvature_reacquired"] is False
    assert (
        curvature_record.certificate[
            "candidate_cross_new_curvature_acquired"
        ]
        is True
    )
    assert curvature_record.certificate["extension_rule"] == (
        "inherited_predicted_old_block_acquired_candidate_cross_new"
    )


def test_indefinite_predicted_same_ray_metric_requires_exact_reanchor() -> None:
    predicted_frame = build_supported_frame(
        np.asarray([[1.4, 0.2], [0.2, 0.9]], dtype=float),
        config=_config(),
        status=STATUS_PREDICTED,
        source_primitive_ids=("predicted-old-old",),
    )

    with pytest.raises(OuterInformationReanchorRequired) as caught:
        extend_same_ray_frame(
            predicted_frame,
            # These acquired candidate overlaps are incompatible with the
            # inherited modeled old--old block and make the augmented metric
            # materially indefinite.  The accepted structure must be handled
            # by an exact reanchor, never by weakening the shared rank rule.
            active_candidate_gram=np.asarray([[2.0], [0.0]], dtype=float),
            candidate_candidate_gram=np.asarray([[1.0]], dtype=float),
            old_to_new_registry_mapping=(0, 1),
            candidate_positions=(2,),
            config=_config(),
            source_primitive_ids=("candidate-blocks",),
        )

    assert caught.value.reason == (
        "enlarged_same_ray_metric_requires_exact_reanchor"
    )
    telemetry = dict(caught.value.telemetry)
    assert telemetry["factorization_reason"] == (
        "materially_negative_metric_eigenvalue"
    )
    assert telemetry["next_action"] == (
        "discard_predictive_geometry_and_use_exact_metric"
    )
    assert telemetry["source_frame_status"] == STATUS_PREDICTED
    assert telemetry["structural_rollback_performed"] is False

def test_active_prior_exposes_predicted_raw_representatives_and_round_trips() -> None:
    session, _, _, _ = _active_direct_growth_session()
    session.predict_refit(
        physical_state_fingerprint="ray:active-refit",
        theta=(0.22, 0.015, -0.08),
        normalized_strain=np.asarray(
            [[0.01, 0.002, 0.0], [0.002, -0.005, 0.001], [0.0, 0.001, 0.004]],
            dtype=float,
        ),
        skew_connection=np.asarray(
            [[0.0, 0.01, -0.02], [-0.01, 0.0, 0.005], [0.02, -0.005, 0.0]],
            dtype=float,
        ),
        step_scale=0.4,
    )
    state = session.pending
    assert state is not None
    prior = active_outer_geometry_prior(
        state, expected_registry=("a", "candidate", "b")
    )
    O = state.frame.raw_orthonormalizer

    assert prior.geometry_status == STATUS_PREDICTED
    assert prior.orientation_status == STATUS_PREDICTED
    np.testing.assert_allclose(prior.raw_metric, state.frame.raw_metric)
    np.testing.assert_allclose(
        O.T @ prior.raw_gradient, state.gradient_covector, atol=2.0e-12
    )
    assert prior.raw_direct_curvature is not None
    np.testing.assert_allclose(
        O.T @ prior.raw_direct_curvature @ O,
        state.curvature,
        atol=2.0e-12,
    )
    assert not prior.raw_metric.flags.writeable
    assert not prior.raw_gradient.flags.writeable
    assert not prior.raw_direct_curvature.flags.writeable

    restored = ActiveOuterGeometryPrior.from_portable_payload(
        prior.portable_payload()
    )
    assert restored.prior_id == prior.prior_id
    np.testing.assert_allclose(restored.raw_metric, prior.raw_metric)
    np.testing.assert_allclose(
        restored.raw_direct_curvature, prior.raw_direct_curvature
    )
    tampered = deepcopy(prior.portable_payload())
    tampered["raw_metric"][0][0] += 0.01
    with pytest.raises(ValueError, match="fingerprint mismatch"):
        ActiveOuterGeometryPrior.from_portable_payload(tampered)
