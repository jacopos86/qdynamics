@pytest.fixture(scope="module")
def planned_campaign(
    tmp_path_factory: pytest.TempPathFactory,
) -> tuple[Path, Path, dict[str, Any]]:
    repo_root = Path(campaign.__file__).resolve().parents[2]
    campaign_dir = tmp_path_factory.mktemp("fm-sr-source-lock") / "campaign"
    payload = campaign.initialize_campaign(
        repo_root=repo_root,
        campaign_dir=campaign_dir,
    )
    return repo_root, campaign_dir, payload


def _stage(payload: dict[str, Any], stage_id: str) -> dict[str, Any]:
    return next(
        dict(stage)
        for stage in payload["stages"]
        if stage["id"] == stage_id
    )

    planned_campaign: tuple[Path, Path, dict[str, Any]],
) -> None:
    _repo_root, _campaign_dir, payload = planned_campaign

    assert payload["scientific_concurrency"] == 1

        "fm_qbroyd_off_weak_weak",
        "fm_qbroyd_off_intermediate_weak_transfer",
    ]
    assert [stage["kind"] for stage in payload["stages"]] == [
        "archived_sr_anchor",
        "current_sr_parity_anchor",
        "fm_source_locked_variant",
        "fm_cross_regime_policy_transfer",
    ]


def test_semantic_diff_allowlist_is_exact(
    planned_campaign: tuple[Path, Path, dict[str, Any]],
) -> None:
    _repo_root, _campaign_dir, payload = planned_campaign
    live_audit = _stage(payload, "current_sr_parity_anchor")[
        "semantic_diff_audit"
    ]
    fm_audit = _stage(payload, "fm_qbroyd_off_weak_weak")[
        "semantic_diff_audit"
    ]

    operational = {
        "adapt_current_json",
        "adapt_estimator_call_ledger_json",
        "output_json",
    }
    fm_mechanism = {
        "adapt_reoptimization_route",
        "adapt_formal_manifold_route_profile",
        "adapt_formal_manifold_config_json",
    }
    assert set(live_audit["changed_destinations"]) == operational
    assert set(live_audit["allowed_destinations"]) == operational
    assert live_audit["unexpected_changed_destinations"] == []
    assert live_audit["non_swept_settings_diff_empty"] is True
    assert set(fm_audit["changed_destinations"]) == operational | fm_mechanism
    assert set(fm_audit["allowed_destinations"]) == operational | fm_mechanism
    assert fm_audit["unexpected_changed_destinations"] == []
    assert fm_audit["non_swept_settings_diff_empty"] is True

def test_immutable_source_anchor_passes_self_parity(
    planned_campaign: tuple[Path, Path, dict[str, Any]],
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    repo_root, _campaign_dir, _payload = planned_campaign
    source_result = (repo_root / campaign.SOURCE_RESULT).resolve()
    original_read_json = campaign._read_json
    cache: dict[Path, dict[str, Any]] = {}

    def memoized_read_json(path: Path) -> dict[str, Any]:
        resolved = Path(path).resolve()
        if resolved not in cache:
            cache[resolved] = original_read_json(resolved)
        return cache[resolved]

    # The immutable result is large.  Cache the read so self-parity validates
    # the real artifact without needlessly parsing the same bytes twice.
    monkeypatch.setattr(campaign, "_read_json", memoized_read_json)
    validation = campaign._validate_anchor_against(
        source_result,
        source_result,
        role="immutable_source_self_parity",
    )

    assert validation["status"] == "pass"
    assert validation["blockers"] == []
    assert validation["energy_delta"] == 0.0
    assert validation["abs_delta_e_delta"] == 0.0
    assert validation["history_rounds"] == 30


def test_anchor_rejects_nonfinite_energy_even_when_comparison_is_nan(
    tmp_path: Path,
    monkeypatch: pytest.MonkeyPatch,
) -> None:
    reference = {
        "adapt_vqe": {
            "energy": -1.0,
            "abs_delta_e": 0.0,
            "operators": ["op"],
            "history": [],
        }
    }
    observed = json.loads(json.dumps(reference))
    observed["adapt_vqe"]["energy"] = float("nan")
    reference_path = tmp_path / "reference.json"
    observed_path = tmp_path / "observed.json"
    reference_path.write_text(json.dumps(reference), encoding="utf-8")
    observed_path.write_text(json.dumps(observed), encoding="utf-8")
    monkeypatch.setattr(
        campaign,
        "_source_selector_contract_blockers",
        lambda _result: [],
    )

    with pytest.raises(ValueError, match="nonfinite_energy_or_error"):
        campaign._validate_anchor_against(
            observed_path,
            reference_path,
            role="nonfinite_guard",
        )

def test_fm_validator_rejects_unsuccessful_empty_terminal_result(
    tmp_path: Path,
) -> None:
    result_path = tmp_path / "result.json"
    current_path = tmp_path / "current.json"
    result_path.write_text(
        json.dumps(
            {
                "settings": {
                    "adapt_formal_manifold_config": campaign._formal_config_payload()
                },
                "adapt_vqe": {
                    "success": False,
                    "energy": campaign.WEAK_WEAK["exact_energy"],
                    "abs_delta_e": 0.0,
                    "operators": [],
                    "history": [],
                },
            }
        ),
        encoding="utf-8",
    )
    current_path.write_text(
        json.dumps({"settings": {}, "adapt_vqe": {}}),
        encoding="utf-8",
    )

    with pytest.raises(ValueError, match="result:success"):
        campaign._validate_fm_result(
            result_path,
            current_path,
            regime=campaign.WEAK_WEAK,
            expected_horizon=1,
        )
