def test_phase2_geometry_payload_is_permuted_into_phase1_anchor_order() -> None:
    payload = {
        "active_indices": [1, 0],
        "G_AA": [[11.0, 10.0], [10.0, 0.0]],
        "G_AB": [101.0, 100.0],
        "H_AA": [[21.0, 20.0], [20.0, 2.0]],
        "H_AB": [201.0, 200.0],
        "g_A": [301.0, 300.0],
    }

    aligned = _phase2_geometry_payload_in_anchor_order(
        payload,
        anchor_active_indices=(0, 1),
    )

    assert aligned["active_indices"] == [0, 1]
    assert aligned["G_AA"] == [[0.0, 10.0], [10.0, 11.0]]
    assert aligned["G_AB"] == [100.0, 101.0]
    assert aligned["H_AA"] == [[2.0, 20.0], [20.0, 21.0]]
    assert aligned["H_AB"] == [200.0, 201.0]
    assert aligned["g_A"] == [300.0, 301.0]
    assert aligned["formal_phase2_active_coordinate_permutation"] == [1, 0]
    assert aligned["formal_phase2_active_coordinate_permutation_applied"] is True

) -> None:

        [
            "--L",
            "2",
            "--skip-pdf",

        SR_ROUTE_PROFILE_NO_PRUNE_SYMMETRIC_COST_V1

        SR_ROUTE_PROFILE_NO_PRUNE_SYMMETRIC_COST_V1

    assert normalized.adapt_formal_manifold_route_profile == route_profile
    for field, expected in expected_contract["execution_settings"].items():
        if field == "adapt_formal_manifold_route_profile":
            continue
