def _term(word: str, coeff: complex = 1.0) -> SerializedPauliExpansionTerm:
    coeff_c = complex(coeff)
    return SerializedPauliExpansionTerm(
        pauli_exyz=word,
        coeff_re=float(coeff_c.real),
        coeff_im=float(coeff_c.imag),
        nq=len(word),
    )


def _exp(key: str, label: str, words: list[str]) -> GeneratorAlgebraicExpansion:
    terms = tuple(_term(word) for word in words)
    support = sorted({q for term in terms for q in support_qubits_from_pauli_word(term.pauli_exyz)})
    return GeneratorAlgebraicExpansion(
        key=key,
        label=label,
        generator_id=key,
        terms=terms,
        support_qubits=tuple(support),
        exactness=EXACTNESS_EXACT,
        source="test",
    )

def test_exact_composite_commutation_uses_full_commutator_not_pairwise_terms() -> None:
    lhs = _exp("a", "A", ["x", "y"])
    rhs = _exp("b", "B", ["x", "y"])

    pair = build_pair_metadata(lhs, rhs)

    assert commutes is True

def test_registry_and_polynomial_sources_build_exact_expansion_index() -> None:
    registry = {
        "reg_op": {
            "candidate_label": "reg_op",
            "generator_id": "gen:reg",
            "support_qubits": [0],
            "compile_metadata": {
                "serialized_terms_exyz": [
                    {"pauli_exyz": "ez", "coeff_re": 2.0, "coeff_im": 0.0, "nq": 2}
                ]
            },
        }
    }
    idx = build_exact_expansion_index(registry_by_label=registry)
    assert idx.resolve_key("reg_op") == "gen:reg"
    assert idx.expansions_by_key["gen:reg"].source == "registry_serialized_terms"

    poly = PauliPolynomial("JW", [PauliTerm(2, ps="xe", pc=1.0)])
    term = SimpleNamespace(label="poly_op", polynomial=poly)
    idx2 = build_exact_expansion_index(pool=[term], registry_by_label={}, allow_polynomial_source=True)
    exp = idx2.expansions_by_key["poly_op"]
    assert exp.source == "ansatz_term_polynomial"
    assert exp.support_qubits == (1,)

def test_algebraic_lane_quota_pressure_budgets_preserve_or_relax_lane_coverage() -> None:
    records = [
        {"candidate_pool_index": 0, "position_id": 0, "algebraic_lane": LANE_FLAT, "simple_score": 10.0},
        {"candidate_pool_index": 1, "position_id": 0, "algebraic_lane": LANE_FLAT, "simple_score": 9.0},
        {"candidate_pool_index": 2, "position_id": 0, "algebraic_lane": LANE_FLAT, "simple_score": 8.0},
        {"candidate_pool_index": 3, "position_id": 0, "algebraic_lane": LANE_FLAT, "simple_score": 7.0},
        {"candidate_pool_index": 4, "position_id": 0, "algebraic_lane": LANE_CURV, "simple_score": 6.0},
        {"candidate_pool_index": 5, "position_id": 0, "algebraic_lane": LANE_DISJ, "simple_score": 5.0},
        {"candidate_pool_index": 6, "position_id": 0, "algebraic_lane": LANE_MIX, "simple_score": 4.0},
    ]

    coverage = algebraic_lane_quota_pressure_budgets(
        records,
        cap=4,
        score_key="simple_score",
        threshold=0.0,
        pressure=1.0,
    )
    assert sum(coverage.values()) == 4
    assert all(coverage[lane] == 1 for lane in (LANE_FLAT, LANE_CURV, LANE_DISJ, LANE_MIX))

    relaxed = algebraic_lane_quota_pressure_budgets(
        records,
        cap=4,
        score_key="simple_score",
        threshold=0.0,
        pressure=0.25,
    )
    assert sum(relaxed.values()) == 4
    assert relaxed[LANE_FLAT] == 4
    assert all(relaxed[lane] == 0 for lane in (LANE_CURV, LANE_DISJ, LANE_MIX))
    for lane in (LANE_FLAT, LANE_CURV, LANE_DISJ, LANE_MIX):
        available = sum(1 for row in records if row["algebraic_lane"] == lane)
        assert relaxed[lane] <= available

    tail_ranked = algebraic_lane_quota_pressure_budgets(
        [
            {"candidate_pool_index": 10, "position_id": 0, "algebraic_lane": LANE_FLAT, "simple_score": 10.0},
            {"candidate_pool_index": 11, "position_id": 0, "algebraic_lane": LANE_FLAT, "simple_score": 1.0},
            {"candidate_pool_index": 12, "position_id": 0, "algebraic_lane": LANE_CURV, "simple_score": 9.0},
            {"candidate_pool_index": 13, "position_id": 0, "algebraic_lane": LANE_DISJ, "simple_score": 8.0},
        ],
        cap=3,
        score_key="simple_score",
        threshold=0.0,
        pressure=0.25,
    )
    assert tail_ranked[LANE_FLAT] == 1
    assert tail_ranked[LANE_CURV] == 1
    assert tail_ranked[LANE_DISJ] == 1
    assert tail_ranked[LANE_MIX] == 0
